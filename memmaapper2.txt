from ghidra.program.model.symbol import SourceType
from ghidra.util.task import TaskMonitor

# Define the Sega/SSFII Hardware Mapper Registers (Motorola 68000 addresses)
MAPPER_REGISTERS = {
    0xA130F1: "SSF2_BANK_SLOT1_080000",
    0xA130F3: "SSF2_BANK_SLOT2_100000",
    0xA130F5: "SSF2_BANK_SLOT3_180000",
    0xA130F7: "SSF2_BANK_SLOT4_200000",
    0xA130F9: "SSF2_BANK_SLOT5_280000",
    0xA130FB: "SSF2_BANK_SLOT6_300000",
    0xA130FD: "SSF2_BANK_SLOT7_380000"
}

def analyze_mapper_writes():
    program = getCurrentProgram()
    listing = program.getListing()
    
    # Label the hardware registers first if they aren't already
    symbol_table = program.getSymbolTable()
    for addr_val, reg_name in MAPPER_REGISTERS.items():
        addr = toAddr(addr_val)
        if symbol_table.getPrimarySymbol(addr) is None:
            symbol_table.createLabel(addr, reg_name, SourceType.USER_DEFINED)
            print("Labeled hardware register: {} at {}".format(reg_name, addr))

    # Loop through all instructions in the ROM
    print("Scanning code for dynamic bank-switching triggers...")
    instruction = listing.getInstructions(toAddr(0x000000), True)
    
    while instruction.hasNext() and not monitor.isCancelled():
        ins = instruction.next()
        mnemonic = ins.getMnemonicString()
        
        # We look for instructions moving data into memory (like 'move.b' or 'movei')
        if "move" in mnemonic.lower():
            num_operands = ins.getNumOperands()
            for i in range(num_operands):
                op_type = ins.getOperandType(i)
                
                # Check if the instruction targets our hardware registers
                # Look for absolute addresses or references
                op_address = ins.getAddress(i)
                if op_address and op_address.getOffset() in MAPPER_REGISTERS:
                    reg_addr = op_address.getOffset()
                    reg_name = MAPPER_REGISTERS[reg_addr]
                    
                    # Try to grab the immediate value being written (the Bank Number)
                    # On M68k, usually operand 0 is the source (value) and operand 1 is destination (register)
                    bank_val = "UNKNOWN"
                    if num_operands > 1:
                        # Extract the literal value being loaded into the register
                        bank_val = ins.getDefaultOperandRepresentation(0)
                    
                    # Calculate the physical ROM offset being exposed
                    # Each bank index represents a 512KB (0x80000) chunk of the 40Mb file
                    try:
                        bank_int = int(bank_val.replace('#', '').replace('$', '0x'), 16)
                        rom_offset = bank_int * 0x80000
                        comment_str = "BANK SWITCH: Loading Bank {} (ROM Offset: 0x{:06X}) into slot {}".format(
                            bank_int, rom_offset, reg_name.split('_')[2]
                        )
                    except ValueError:
                        comment_str = "BANK SWITCH: Dynamic write to {} detected.".format(reg_name)
                    
                    # Place a primary code comment right next to the instruction
                    ins.setComment(ins.PLATE_COMMENT, comment_str)
                    print("Found trigger at {}: {}".format(ins.getAddressString(), comment_str))

# Execute the analysis

import os

def split_sonic_delta_rom(rom_path, output_dir="extracted_banks"):
    # The SSFII mapper shifts data in strict 512 KB chunks (0x80000 bytes)
    BANK_SIZE = 0x80000  
    
    # Verify the file exists
    if not os.path.exists(rom_path):
        print(f"Error: Could not find the ROM file at '{rom_path}'")
        print("Please check the filename and try again.")
        return

    # Check the file size to ensure it's the 40-Megabit version
    file_size = os.path.getsize(rom_path)
    file_size_mb = (file_size * 8) / 1000000
    print(f"Opening ROM: {rom_path} ({file_size_mb:.1f} Megabits / {file_size} bytes)")

    # Create the output directory if it doesn't exist
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
        print(f"Created output directory: '{output_dir}/'")

    # Read the ROM and slice it up sequentially
    with open(rom_path, "rb") as rom_file:
        bank_number = 0
        while True:
            # Read exactly 512 KB at a time
            bank_data = rom_file.read(BANK_SIZE)
            
            # Break the loop if we hit the end of the file
            if not bank_data:
                break
                
            # Calculate the hex range for clear labeling
            start_offset = bank_number * BANK_SIZE
            end_offset = start_offset + len(bank_data) - 1
            
            # Generate the file name for this specific chunk
            output_filename = f"bank_{bank_number:02d}_[0x{start_offset:06X}-0x{end_offset:06X}].bin"
            output_path = os.path.join(output_dir, output_filename)
            
            # Write out the isolated bank file
            with open(output_path, "wb") as out_file:
                out_file.write(bank_data)
                
            print(f" -> Successfully extracted Bank {bank_number:02d} -> {output_filename}")
            bank_number += 1

    print("\nExtraction complete! All hardware banks have been cleanly separated.")
    print("You can now cross-reference these files individually inside Ghidra or your hex editor.")

\U0001f3d7\ufe0f Hardware Architecture & Memory Layout

The Sega Genesis CPU (Motorola 68000) has a native direct addressing limit of 32 Megabits (4 Megabytes) for ROM space (0x000000\u2013$3FFFFF). Because Sonic Delta Origins is a 40-Megabit (5 Megabyte) file, it requires a hardware-level bank-switching layout.
Neto repurposed the official commercial layout created by SEGA/Capcom for the 1993 Genesis release of Super Street Fighter II (SSFII).

1. The Bank Slot Architecture

The first 512 KB of the game (Bank 00) is permanently locked at the beginning of the memory space. The rest of the ROM address space is divided into seven dynamic 512 KB "Slots."

Memory Window	Slot Name	Hardware Register	Purpose / Behavior
0x000000\u20130x07FFFF	Bank 00 (Fixed)	None (Locked)	Main engine vectors, boot routines, and menu selectors.
0x080000\u20130x0FFFFF	Slot 1	0xA130F1	Dynamically swaps active game code blocks.
0x100000\u20130x17FFFF	Slot 2	0xA130F3	Dynamically swaps active game code blocks.

The Z80 audio coprocessor lives inside its own small memory bubble ($0000\u2013$7FFF). To play high-quality sound effects like Sega chants or drum kits, it has to read them from the main ROM space using a tiny 32 KB sliding window located at Z80 address $8000.

# Sega Genesis Engine Kosinski Decompression Handler Macro
def decompress_kosinski(compressed_data):
    """
    Pure Python routine to decode standard Sega Kosinski (LZSS variation) streams
    found natively inside the extracted Sonic level graphics banks.
    """
    input_bytes = list(compressed_data)
    output = bytearray()
    src = 0
    
    # Context window values tracking
    descriptor = 0
    bit_count = 0
    
    def get_match_bit():
        nonlocal descriptor, bit_count, src
        if bit_count == 0:
            # Read new 16-bit status window descriptor
            if src + 1 >= len(input_bytes): return 0
            descriptor = input_bytes[src] | (input_bytes[src+1] << 8)
            src += 2
            bit_count = 16
        bit = descriptor & 1
        descriptor >>= 1
        bit_count -= 1
        return bit

    try:
        while src < len(input_bytes):
            if get_match_bit():
                # Direct uncompressed byte transfer literal copy
                output.append(input_bytes[src])
                src += 1
            else:
                # History lookup matching sliding window flag
                if get_match_bit():
                    # 2-byte distance format parameter string
                    low = input_bytes[src]
                    high = input_bytes[src+1]
                    src += 2
                    
                    # Distance offset math
                    offset = ((high & 0xF8) << 5) | low
                    offset = -((offset ^ 0x1FFF) + 1)
                    count = high & 0x07
                    
                    if count == 0:
                        count = input_bytes[src] + 1
                        src += 1
                    else:
                        count += 1
                else:
                    # Short window offset allocation handling routine
                    bit1 = get_match_bit()
                    bit2 = get_match_bit()
                    count = (bit1 << 1) | bit2
                    offset = -input_bytes[src] - 1
                    src += 1
                    count += 1
                
                # Copy pattern bytes from history window
                for _ in range(count + 1):
                    target_idx = len(output) + offset
                    output.append(output[target_idx])
    except IndexError:
        pass # Stream boundary terminal breakout safety
        
    return bytes(output)

def extract_genesis_palette(rom_chunk_path, start_offset, num_colors=16):
    """
    Extracts raw 9-bit Mega Drive hardware palettes (encoded as 16-bit big-endian words)
    and formats them cleanly into modern RGB system hex values.
    """
    extracted_palette = []
    
    with open(rom_chunk_path, "rb") as f:
        # Seek straight to the offset flagged in your disassembler
        f.seek(start_offset)
        
        for _ in range(num_colors):
            color_bytes = f.read(2)
            if len(color_bytes) < 2:
                break
                
            # Combine bytes to recreate the big-endian 16-bit word
            raw_val = (color_bytes[0] << 8) | color_bytes[1]

            green = (raw_val & 0x00E0) >> 5

            # Scale the 0-7 Genesis steps up to standard 0-255 RGB channels
            # Multiply by 36 maps out the steps uniformly (7 * 36 = 252 ~ #FC)
            extracted_palette.append(f"#{red*36:02X}{green*36:02X}{blue*36:02X}")
            
    return extracted_palette

Part 2: Automated Asset Verification Tool (verify_assets.py)

When building massive level sets or randomizers, you must ensure your extracted or procedurally generated asset binaries don't cross boundaries. If a level's art block accidentally exceeds its allocated 512 KB chunk size (0x80000 bytes), vasm will throw compile-time padding errors.

python
import os
import sys

def verify_bank_boundaries(asset_dir="src/assets"):
    """
    Scans asset directories to ensure graphics, level grids, and audio chunks
    do not overflow the hardware bank boundary constraints of the SSFII mapper.
    """
    MAX_BANK_SIZE = 0x80000  # Strict 512 KB Limit per segment
    errors_detected = 0

    print("=" * 65)
    print("      UNIVERSAL FRAMEWORK: HARDWARE BOUNDARY VERIFICATION")
    print("=" * 65)

    if not os.path.exists(asset_dir):
        print(f"[!] Warning: Asset staging directory '{asset_dir}' not found.")
        print("    Ensure your asset folders are populated. Aborting scan.")
        return

    # Crawl through subdirectories checking all raw binaries (.bin, .kos, .pcm)
    for root, _, files in os.walk(asset_dir):
        for file in files:
            if file.endswith(('.bin', '.kos', '.pcm', '.dat')):
                file_path = os.path.join(root, file)
                size_bytes = os.path.getsize(file_path)
                
                # Check file size parameters relative to physical banking limits
                if size_bytes > MAX_BANK_SIZE:
                    print(f"[CRITICAL ERROR] '{file}' is too large for a single bank!")
                    print(f"                 Size: {size_bytes} / {MAX_BANK_SIZE} bytes.")
                    print(f"                 Overflow by: {size_bytes - MAX_BANK_SIZE} bytes.")
                    errors_detected += 1
                else:
                    percentage = (size_bytes / MAX_BANK_SIZE) * 100
                    print(f"[OK] {file:<25} Size: {size_bytes:6d} bytes ({percentage:5.1f}% of Bank capacity)")

    print("-" * 65)
    if errors_detected > 0:
        print(f"[!] Verification FAILED. {errors_detected} assets will cause compilation overflows.")
        print("    Action required: Defragment or split the oversized files listed above.")
        sys.exit(1)
    else:
        print("[+] Verification SUCCESS. All asset bounds conform to SSFII hardware specs.")
        print("=" * 65)

if __name__ == "__main__":
    # Point this to your asset payload subfolder path
    verify_bank_boundaries()
