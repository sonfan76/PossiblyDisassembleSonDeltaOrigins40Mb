
dynamic analysis is needed
Dynamic runtime analysis bypasses static pointer obfuscation by logging the exact CPU register states and memory addresses at the moment a hardware bank switch executes. Because the Motorola 68000 must write directly to a mapper latch (such as the standard Sega SSF2 mapper range at $A130F1–$A130FF or custom registers at $220000–$22000F) to swap 512KB banks, trapping that write exposes the exact jump targets and bank numbers in real time.
1. MAME / Mesen Lua Dynamic Watchpoint Logger
This Lua script hooks into the emulator's memory bus to catch every write to suspected bank-switching ranges, logs the full M68K register state (PC, d0–d7, a0–a6), and writes the output to a file (bank_trace.log) so you can map which bank corresponds to each level or routine.
lua
-- =============================================================================
-- M68K DYNAMIC BANK-SWITCH LOGGER (MAME / Mesen Lua API)
-- Captures PC, full CPU register context, and bank values on mapper writes
-- =============================================================================

local log_file = io.open("bank_trace.log", "w")
local seen_switches = {}

-- Common Sega Genesis / MD Mapper Register Ranges:
-- 1) Standard SSF2 Mapper: 0xA130F1 to 0xA130FF (odd bytes)
-- 2) Custom / Extended Mapper Range: 0x220000 to 0x22000F
local WATCH_RANGES = {
    { start_addr = 0xA130F0, end_addr = 0xA130FF },
    { start_addr = 0x220000, end_addr = 0x22000F }
}

local function dump_m68k_state(pc, addr, val, size)
    local key = string.format("%06X_%06X_%04X", pc, addr, val)
    
    -- Deduplicate repeated identical switches in tight loops; remove check to log every hit
    if seen_switches[key] then return end
    seen_switches[key] = true

    local regs = string.format(
        "D0=%08X D1=%08X D2=%08X D3=%08X D4=%08X D5=%08X D6=%08X D7=%08X\n" ..
        "A0=%08X A1=%08X A2=%08X A3=%08X A4=%08X A5=%08X A6=%08X A7=%08X",
        emu:get_reg("D0"), emu:get_reg("D1"), emu:get_reg("D2"), emu:get_reg("D3"),
        emu:get_reg("D4"), emu:get_reg("D5"), emu:get_reg("D6"), emu:get_reg("D7"),
        emu:get_reg("A0"), emu:get_reg("A1"), emu:get_reg("A2"), emu:get_reg("A3"),
        emu:get_reg("A4"), emu:get_reg("A5"), emu:get_reg("A6"), emu:get_reg("A7")
    )

    local entry = string.format(
        "[BANK SWITCH] PC=$%06X -> Wrote $%04X (%d-bit) to $%06X\n%s\n" ..
        "------------------------------------------------------------------------\n",
        pc, val, size * 8, addr, regs
    )

    print(entry)
    if log_file then
        log_file:write(entry)
        log_file:flush()
    end
end

-- Register write callbacks across all target mapper windows
for _, range in ipairs(WATCH_RANGES) do
    for addr = range.start_addr, range.end_addr, 2 do
        emu:add_watchpoint(addr, 2, "w", function()
            local pc = emu:get_reg("PC")
            local target = emu:get_watchpoint_address()
            local val = emu:read_word(target)
            dump_m68k_state(pc, target, val, 2)
        end)
    end
end

print("[DYNAMIC TRACE] Watchpoints active. Play through level transitions to log bank swaps.")

2. Interactive Commands When a Watchpoint Hits
When the ROM writes to a bank register during a level transition or asset load, BlastEm will pause execution and drop you into the (blastem) prompt [1.14]. Run these commands to inspect the exact state:
    • p (Print Registers): Displays all data registers (d0–d7), address registers (a0–a7), the current Program Counter (pc), and the Status Register (sr) [1.14].
    • bt (Backtrace): Walks the stack frame via a7 to show the exact chain of jsr / bsr return addresses that called the bank-switching routine [1.14].
    • x/16x $a7 (Examine Stack): Dumps the top 16 longwords on the stack so you can see the raw return pointer at (a7) and any pushed parameters.
    • disasm (Disassemble): Disassembles the instructions immediately surrounding the current pc so you can see the exact move.b or move.w instruction that wrote the bank index and the jump/pointer load that follows it.
    • c (Continue): Resumes execution until the next bank swap occurs.

any way to automate it?
You can fully automate the entire pipeline—capturing bank swaps in BlastEm, resolving every cross-bank pointer into physical file offsets, building the Ghidra memory overlays, and exporting clean vasm-compatible .asm files—using a two-stage automation chain.

Stage 1: Headless BlastEm Trace Collector (collect_trace.py)
Instead of manually typing p, disasm, and c every time BlastEm hits a watchpoint, this Python wrapper launches blastem -d in the background, automatically feeds those inspection commands every time a mapper write triggers, and saves a clean JSON map (resolved_switches.json) pairing every caller PC with its target bank and physical ROM offset.
python
#!/usr/bin/env python3
import subprocess
import re
import json
import sys

ROM_FILE = "sonic_delta_origins.bin"
OUTPUT_JSON = "resolved_switches.json"

# Standard SSF2 mapper registers -> CPU swap window base
SSF2_WINDOWS = {
    0xA130F3: 0x080000,
    0xA130F5: 0x100000,
    0xA130F7: 0x180000,
    0xA130F9: 0x200000,
    0xA130FB: 0x280000,
    0xA130FD: 0x300000,
    0xA130FF: 0x380000,
}

def run_blastem_trace(rom_path):
    proc = subprocess.Popen(
        ["blastem", "-d", rom_path],
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        text=True,
        bufsize=1
    )

    # Set watchpoints across SSF2 and custom $220000 ranges, then continue
    init_cmds = [f"watch ${reg:06X}" for reg in SSF2_WINDOWS] + ["watch $220000", "c"]
    proc.stdin.write("\n".join(init_cmds) + "\n")
    proc.stdin.flush()

    switches = {}
    current_block = []

    print("[TRACE] Running BlastEm. Play through zones/menus, then close emulator window to finish.")

    for line in proc.stdout:
        line_clean = line.strip()
        current_block.append(line_clean)

        # Detect BlastEm watchpoint hit prompt
        if "Watchpoint" in line_clean or "(blastem)" in line_clean:
            if "Watchpoint" in line_clean:
                proc.stdin.write("p\ndisasm\nc\n")
                proc.stdin.flush()

            blob = "\n".join(current_block)
            pc_match = re.search(r"pc:\s*([0-9a-fA-F]+)", blob)
            wp_match = re.search(r"Watchpoint.*?([0-9a-fA-F]{6})", blob)
            d0_match = re.search(r"d0:\s*([0-9a-fA-F]+)", blob)
            a0_match = re.search(r"a0:\s*([0-9a-fA-F]+)", blob)

            if pc_match and wp_match:
                pc = int(pc_match.group(1), 16)
                reg = int(wp_match.group(1), 16)
                bank = int(d0_match.group(1), 16) & 0xFF if d0_match else 0
                a0 = int(a0_match.group(1), 16) if a0_match else 0

                win_base = SSF2_WINDOWS.get(reg, 0x380000)
                phys_offset = ((a0 - win_base) + (bank * 0x80000)) if a0 >= win_base else (bank * 0x80000)

                key = f"{pc:06X}_{reg:06X}_{bank:02X}"
                if key not in switches:
                    switches[key] = {
                        "caller_pc": f"0x{pc:06X}",
                        "mapper_reg": f"0x{reg:06X}",
                        "bank_index": bank,
                        "window_base": f"0x{win_base:06X}",
                        "sample_a0": f"0x{a0:06X}",
                        "physical_rom_offset": f"0x{phys_offset:06X}"
                    }
                    print(f"[CAPTURED] PC=${pc:06X} -> Reg=${reg:06X} Bank={bank:02d} (Phys=0x{phys_offset:06X})")
                current_block.clear()

    with open(OUTPUT_JSON, "w") as f:
        json.dump(list(switches.values()), f, indent=2)
    print(f"[DONE] Saved {len(switches)} unique bank switches to {OUTPUT_JSON}")

if __name__ == "__main__":
    target = sys.argv[1] if len(sys.argv) > 1 else ROM_FILE
    run_blastem_trace(target)
Use code with caution.

Stage 2: Ghidra Headless Overlay Builder & vasm Exporter (ghidra_auto_bank.py)
Once you have resolved_switches.json (or even directly from the raw 40Mbit .bin), run this Ghidra Python script via Ghidra's Script Manager (or headless CLI via analyzeHeadless). It automatically:
    1. Slices every 512KB bank above 4MB (0x400000) into a named Ghidra Overlay Block (BANK_08, BANK_09, etc.).
    2. Reads resolved_switches.json to attach cross-bank references (Add/Edit Reference) and bookmark comments directly at each caller PC.

# Ghidra Jython Script: Auto-Map 512KB Overlays, Apply Trace Refs, & Export vasm ASM
# @category SegaGenesis.Automation

import os
import json
from ghidra.app.cmd.disassemble import DisassembleCommand
from ghidra.program.model.symbol import SourceType, RefType

BANK_SIZE = 0x80000          # 512KB per bank
BASE_ROM_LIMIT = 0x400000    # First 4MB (32Mbit) mapped linearly
DEFAULT_WINDOW = 0x380000    # Default swap window ($380000-$3FFFFF)
OUTPUT_DIR = os.path.expanduser("~/sonic_delta_vasm_export")
TRACE_JSON = os.path.expanduser("~/resolved_switches.json")

mem = currentProgram.getMemory()
listing = currentProgram.getListing()
ref_mgr = currentProgram.getReferenceManager()
addr_factory = currentProgram.getAddressFactory()

# 1. Locate raw FileBytes from the imported 40Mbit ROM
all_file_bytes = mem.getAllFileBytes()
if not all_file_bytes:
    print("[ERROR] No FileBytes found in current program.")
    exit()

rom_bytes = all_file_bytes[0]
total_size = rom_bytes.getSize()
num_banks = total_size // BANK_SIZE
print("[GHIDRA] Total ROM size: {} bytes ({} x 512KB banks)".format(total_size, num_banks))

# 2. Create Overlay Blocks for Banks 8+ (everything above 4MB)
for bank_idx in range(8, num_banks):
    block_name = "BANK_{:02d}".format(bank_idx)
    file_offset = bank_idx * BANK_SIZE
    if mem.getBlock(block_name) is None:
        start_addr = addr_factory.getDefaultAddressSpace().getAddress(DEFAULT_WINDOW)
        block = mem.createInitializedBlock(
            block_name, start_addr, rom_bytes, file_offset, BANK_SIZE, True
        )
        block.setRead(True)
        block.setWrite(False)
        block.setExecute(True)
        print("[OVERLAY] Created {} -> CPU ${:06X} (File Offset 0x{:06X})".format(
            block_name, DEFAULT_WINDOW, file_offset))

# 3. Apply Dynamic Trace References from BlastEm JSON (if present)
if os.path.exists(TRACE_JSON):
    with open(TRACE_JSON, "r") as f:
        switches = json.load(f)
    for sw in switches:
        pc_val = int(sw["caller_pc"], 16)
        bank_idx = sw["bank_index"]
        target_a0 = int(sw["sample_a0"], 16)
        caller_addr = addr_factory.getDefaultAddressSpace().getAddress(pc_val)

        # Annotate caller PC with exact bank switch metadata
        listing.setComment(
            caller_addr,
            listing.EOL_COMMENT,
            "AUTO-TRACE: Swaps Bank {:02d} via {} (Sample Target=0x{:06X})".format(
                bank_idx, sw["mapper_reg"], target_a0
            )
        )

        # Add cross-bank reference into the corresponding Overlay Address Space
        overlay_space = addr_factory.getAddressSpace("BANK_{:02d}".format(bank_idx))
        if overlay_space and target_a0 >= DEFAULT_WINDOW:
            dest_addr = overlay_space.getAddress(target_a0)
            ref_mgr.addMemoryReference(
                caller_addr, dest_addr, RefType.DATA, SourceType.USER_DEFINED, 0
            )
            DisassembleCommand(dest_addr, None, True).applyTo(currentProgram)

# 4. Export every bank block into clean vasm-compatible .asm files
if not os.path.exists(OUTPUT_DIR):
    os.makedirs(OUTPUT_DIR)

for block in mem.getBlocks():
    if not block.isInitialized():
        continue
    out_path = os.path.join(OUTPUT_DIR, "{}.asm".format(block.getName()))
    with open(out_path, "w") as asm_out:
        asm_out.write("; ============================================================================\n")
        asm_out.write("; AUTO-EXPORTED BANK: {} (Start: ${:06X}, Size: ${:X})\n".format(
            block.getName(), block.getStart().getOffset(), block.getSize()))
        asm_out.write("; Assembler Target: vasm -Fm68k -Fbin\n")
        asm_out.write("; ============================================================================\n")
        asm_out.write("\torg\t${:06X}\n\n".format(block.getStart().getOffset()))

        inst_iter = listing.getInstructions(block.getStart(), True)
        while inst_iter.hasNext():
            inst = inst_iter.next()
            if not block.contains(inst.getAddress()):
                break
            addr_off = inst.getAddress().getOffset()
            asm_out.write("loc_{:06X}:\t{}\n".format(addr_off, inst.toString()))

    print("[EXPORT] Wrote {}".format(out_path))

1. Upgraded BlastEm Disassembly & Operand Parser (collect_trace.py)
Rather than guessing that the bank number is always in d0, this version parses the disasm output emitted by BlastEm at each watchpoint break. It inspects the exact instruction that wrote to $A130Fx (or $220000) and resolves immediate constants (#$09), data registers (d0–d7), or memory addresses (($FFFF8000).w).
python
#!/usr/bin/env python3
import subprocess
import re
import json
import sys

ROM_FILE = "sonic_delta_origins.bin"
OUTPUT_JSON = "resolved_switches.json"

SSF2_WINDOWS = {
    0xA130F3: 0x080000,
    0xA130F5: 0x100000,
    0xA130F7: 0x180000,
    0xA130F9: 0x200000,
    0xA130FB: 0x280000,
    0xA130FD: 0x300000,
    0xA130FF: 0x380000,
}

def extract_bank_value(disasm_lines, regs):
    """
    Parses the instruction writing to the mapper register (e.g., 'move.b #$09, $a130ff'
    or 'move.b d1, (a2)') and resolves the exact bank index byte.
    """
    for line in disasm_lines:
        # Match move.b or move.w targeting a mapper address or indirect register
        m = re.search(r"move\.[bw]\s+([^,]+),\s*(.+)", line, re.IGNORECASE)
        if not m:
            continue
        src = m.group(1).strip().lower()

        # Case 1: Immediate hex or decimal value (e.g. #$09 or #9)
        if src.startswith("#$"):
            return int(src[2:], 16) & 0xFF
        if src.startswith("#"):
            return int(src[1:], 10) & 0xFF

        # Case 2: Data register d0-d7
        if re.fullmatch(r"d[0-7]", src) and src in regs:
            return regs[src] & 0xFF

        # Case 3: Indirect address register (a0)-(a6)
        ind = re.fullmatch(r"\((a[0-6])\)\+?", src)
        if ind and ind.group(1) in regs:
            return regs[ind.group(1)]

    # Fallback to d0 if operand format was complex
    return regs.get("d0", 0) & 0xFF

            if pc_match and wp_match:

                reg = int(wp_match.group(1), 16)

1. Ghidra-to-vasm Syntax Normalizer (format_for_vasm.py)
Ghidra's default inst.toString() output uses C-style 0x hex literals, non-standard indirect displacement formats ((0x10,A0) instead of $10(a0)), and omits raw binary bytes for data regions (such as compressed Nemesis/Kosinski art tiles, level layouts, and sound drivers).

python
#!/usr/bin/env python3
import re
import glob
import os

def convert_ghidra_line(line):
    # Preserve comments and org directives
    stripped = line.strip()
    if not stripped or stripped.startswith(";") or stripped.startswith("org"):
        return line

    # Split optional label prefix (e.g. 'loc_000200:\tmove.l 0x0,d0')
    label_part = ""
    inst_part = line
    if ":" in line and not line.lstrip().startswith(";"):
        idx = line.index(":")
        label_part = line[:idx + 1]
        inst_part = line[idx + 1:]

    # 1. Convert C-style hex '0x1A2B' -> Motorola '$1A2B'
    inst_part = re.sub(r"\b0x([0-9a-fA-F]+)\b", r"$\1", inst_part)

    # 2. Convert Ghidra indirect displacement '($10,A0)' -> '$10(a0)'
    inst_part = re.sub(
        r"\(\s*(-?\$?[0-9a-fA-F]+)\s*,\s*([aAdD][0-7]|sp|SP|pc|PC)\s*\)",
        r"\1(\2)",
        inst_part
    )

    # 3. Convert Ghidra indexed indirect '($10,A0,D0.w)' -> '$10(a0,d0.w)'
    inst_part = re.sub(
        r"\(\s*(-?\$?[0-9a-fA-F]+)\s*,\s*([aAdD][0-7]|sp|SP|pc|PC)\s*,\s*([aAdD][0-7]\.[wlWL])\s*\)",
        r"\1(\2,\3)",
        inst_part
    )

    # 4. Normalize '.b'/'.w'/'.l' size suffixes and lowercase registers for vasm
    inst_part = re.sub(r"\b([AD][0-7]|SP|PC)\b", lambda m: m.group(1).lower(), inst_part)

    # 5. Fix absolute short/long suffixes if Ghidra emits '.w'/'.l' without parentheses
    inst_part = re.sub(r"(\$[0-9a-fA-F]{1,4})\.w\b", r"(\1).w", inst_part)
    inst_part = re.sub(r"(\$[0-9a-fA-F]{5,8})\.l\b", r"(\1).l", inst_part)

    return f"{label_part}{inst_part}"

def process_all_banks(asm_dir="."):
    asm_files = sorted(glob.glob(os.path.join(asm_dir, "BANK_*.asm")))
    if not asm_files:
        print("[ERROR] No BANK_*.asm files found to format.")
        return

    for path in asm_files:
        with open(path, "r") as f:
            raw_lines = f.readlines()

        clean_lines = [convert_ghidra_line(l) for l in raw_lines]

        with open(path, "w") as f:
            f.writelines(clean_lines)

        print(f"[FORMAT] Normalized {os.path.basename(path)} ({len(clean_lines)} lines) -> vasm Motorola syntax")

if __name__ == "__main__":
    process_all_banks(".")

1. Two-Pass Symbolic Label Resolver (format_for_vasm.py)
Raw hardcoded hex addresses in branches (beq.s $00024A), subroutines (jsr $001420), and PC-relative loads (lea $000500(pc),a1) break the moment you add or remove a single instruction. This upgraded formatter runs two passes over every .asm file:
    • Pass 1 (Symbol Table Extraction): Scans every line in the bank and indexes all defined loc_XXXXXX: and dat_XXXXXX: addresses.
    • Pass 2 (Operand Symbol Substitution): Normalizes Ghidra's Motorola syntax and rewrites any target address matching a known label inside that bank ($00024A → loc_00024A, $000500(pc) → dat_000500(pc)), while leaving external hardware I/O ($00C00004, $00A130FF) and RAM variables ($FFFF8000) untouched.
python
#!/usr/bin/env python3
import re
import glob
import os

# Control-flow and PC-relative mnemonics whose address operands should become symbolic labels
BRANCH_OR_PTR_MNEMONICS = (
    "bra", "bsr", "beq", "bne", "bpl", "bmi", "bcc", "bcs", "bvc", "bvs",
    "bge", "bgt", "ble", "blt", "bhi", "bls", "jmp", "jsr", "lea", "pea",
    "dbra", "dbf", "dbeq", "dbne", "dbpl", "dbmi", "dbcc", "dbcs",
    "dbge", "dbgt", "dble", "dblt", "dbhi", "dbls"
)

def normalize_syntax(inst_part):
    # 1. Convert C-style hex '0x1A2B' -> Motorola '$1A2B'
    inst_part = re.sub(r"\b0x([0-9a-fA-F]+)\b", r"$\1", inst_part)

    # 2. Convert Ghidra indirect displacement '($10,A0)' -> '$10(a0)'
    inst_part = re.sub(
        r"\(\s*(-?\$?[0-9a-fA-F]+)\s*,\s*([aAdD][0-7]|sp|SP|pc|PC)\s*\)",
        r"\1(\2)",
        inst_part
    )

    # 3. Convert Ghidra indexed indirect '($10,A0,D0.w)' -> '$10(a0,d0.w)'
    inst_part = re.sub(
        r"\(\s*(-?\$?[0-9a-fA-F]+)\s*,\s*([aAdD][0-7]|sp|SP|pc|PC)\s*,\s*([aAdD][0-7]\.[wlWL])\s*\)",
        r"\1(\2,\3)",
        inst_part
    )

    # 4. Lowercase M68K register names
    inst_part = re.sub(r"\b([AD][0-7]|SP|PC)\b", lambda m: m.group(1).lower(), inst_part)

    # 5. Wrap absolute .w / .l suffixes cleanly for vasm
    inst_part = re.sub(r"(\$[0-9a-fA-F]{1,4})\.w\b", r"(\1).w", inst_part)
    inst_part = re.sub(r"(\$[0-9a-fA-F]{5,8})\.l\b", r"(\1).l", inst_part)
    return inst_part

def process_bank_file(path):
    with open(path, "r") as f:
        raw_lines = f.readlines()

    # PASS 1: Collect every defined loc_XXXXXX and dat_XXXXXX symbol in this bank
    symbol_map = {}
    for line in raw_lines:
        m = re.match(r"^((?:loc|dat)_([0-9a-fA-F]{6})):", line.strip())
        if m:
            full_label = m.group(1)
            addr_int = int(m.group(2), 16)
            symbol_map[addr_int] = full_label

    # PASS 2: Normalize syntax and substitute local hex addresses with symbolic labels
    out_lines = []
    replaced_count = 0

    for line in raw_lines:
        stripped = line.strip()
        if not stripped or stripped.startswith(";") or stripped.startswith("org"):
            out_lines.append(line)
            continue

        label_part = ""
        inst_part = line
        if ":" in line and not line.lstrip().startswith(";"):
            idx = line.index(":")
            label_part = line[:idx + 1]
            inst_part = line[idx + 1:]

        inst_part = normalize_syntax(inst_part)

        # Check if instruction is a branch, jump, or PC-relative pointer load
        op_match = re.match(r"^\s*([a-z]+)(?:\.[bws])?\s+(.+)$", inst_part, re.IGNORECASE)
        if op_match:
            mnemonic = op_match.group(1).lower()
            operands = op_match.group(2)

            if mnemonic in BRANCH_OR_PTR_MNEMONICS:
                def sub_addr(match):
                    nonlocal replaced_count
                    raw_hex = match.group(1)
                    suffix = match.group(2) or ""
                    target_int = int(raw_hex, 16)
                    # Only replace if target exists inside this bank's symbol table
                    if target_int in symbol_map:
                        replaced_count += 1
                        return f"{symbol_map[target_int]}{suffix}"
                    return match.group(0)

                # Replace '$00024A' or '$00024A(pc)' with 'loc_00024A' / 'dat_00024A(pc)'
                operands = re.sub(
                    r"(?<!#)\$([0-9a-fA-F]{4,8})(\(pc\))?",
                    sub_addr,
                    operands,
                    flags=re.IGNORECASE
                )
                inst_part = re.sub(
                    r"^(\s*[a-z]+(?:\.[bws])?\s+).+$",
                    rf"\g<1>{operands}",
                    inst_part,
                    flags=re.IGNORECASE
                )

        out_lines.append(f"{label_part}{inst_part}")

    with open(path, "w") as f:
        f.writelines(out_lines)

    print(f"[SYMBOLIC OK] {os.path.basename(path)}: {len(symbol_map)} labels indexed, {replaced_count} operands resolved")

def process_all_banks(asm_dir="."):
    asm_files = sorted(glob.glob(os.path.join(asm_dir, "BANK_*.asm")))
    if not asm_files:
        print("[ERROR] No BANK_*.asm files found.")
        return
    for path in asm_files:
        process_bank_file(path)

if __name__ == "__main__":
    process_all_banks(".")

def extract_bank_value(disasm_lines, regs):
    \"\"\"
    Parses the instruction writing to the mapper register (e.g., 'move.b #$09, $a130ff'
    or 'move.b d1, (a2)') and resolves the exact bank index byte.
    \"\"\"
    for line in disasm_lines:
        m = re.search(r"move\\.[bw]\\s+([^,]+),\\s*(.+)", line, re.IGNORECASE)
        if not m:
            continue
        src = m.group(1).strip().lower()

        # Case 1: Immediate hex or decimal value (e.g. #$09 or #9)
        if src.startswith("#$"):
            return int(src[2:], 16) & 0xFF
        if src.startswith("#"):
            return int(src[1:], 10) & 0xFF

        # Case 2: Data register d0-d7
        if re.fullmatch(r"d[0-7]", src) and src in regs:
            return regs[src] & 0xFF

        # Case 3: Indirect address register (a0)-(a6)
        ind = re.fullmatch(r"\\((a[0-6])\\)\\+?", src)
        if ind and ind.group(1) in regs:
            return regs[ind.group(1)]

    return regs.get("d0", 0) & 0xFF

files["format_for_vasm.py"] = """#!/usr/bin/env python3
import re
import glob
import os

BRANCH_OR_PTR_MNEMONICS = (
    "bra", "bsr", "beq", "bne", "bpl", "bmi", "bcc", "bcs", "bvc", "bvs",
    "bge", "bgt", "ble", "blt", "bhi", "bls", "jmp", "jsr", "lea", "pea",
    "dbra", "dbf", "dbeq", "dbne", "dbpl", "dbmi", "dbcc", "dbcs",
    "dbge", "dbgt", "dble", "dblt", "dbhi", "dbls"
)

def normalize_syntax(inst_part):
    # 1. Convert C-style hex '0x1A2B' -> Motorola '$1A2B'
    inst_part = re.sub(r"\\b0x([0-9a-fA-F]+)\\b", r"$\\1", inst_part)

    # 2. Convert Ghidra indirect displacement '($10,A0)' -> '$10(a0)'
    inst_part = re.sub(
        r"\\(\\s*(-?\\$?[0-9a-fA-F]+)\\s*,\\s*([aAdD][0-7]|sp|SP|pc|PC)\\s*\\)",
        r"\\1(\\2)",
        inst_part
    )

    # 3. Convert Ghidra indexed indirect '($10,A0,D0.w)' -> '$10(a0,d0.w)'
    inst_part = re.sub(
        r"\\(\\s*(-?\\$?[0-9a-fA-F]+)\\s*,\\s*([aAdD][0-7]|sp|SP|pc|PC)\\s*,\\s*([aAdD][0-7]\\.[wlWL])\\s*\\)",
        r"\\1(\\2,\\3)",
        inst_part
    )

    # 4. Lowercase M68K register names
    inst_part = re.sub(r"\\b([AD][0-7]|SP|PC)\\b", lambda m: m.group(1).lower(), inst_part)

    # 5. Wrap absolute .w / .l suffixes cleanly for vasm
    inst_part = re.sub(r"(\\$[0-9a-fA-F]{1,4})\\.w\\b", r"(\\1).w", inst_part)
    inst_part = re.sub(r"(\\$[0-9a-fA-F]{5,8})\\.l\\b", r"(\\1).l", inst_part)
    return inst_part

def process_bank_file(path):
    with open(path, "r") as f:
        raw_lines = f.readlines()

    # PASS 1: Collect every defined loc_XXXXXX and dat_XXXXXX symbol in this bank
    symbol_map = {}
    for line in raw_lines:
        m = re.match(r"^((?:loc|dat)_([0-9a-fA-F]{6})):", line.strip())
        if m:
            full_label = m.group(1)
            addr_int = int(m.group(2), 16)
            symbol_map[addr_int] = full_label

    # PASS 2: Normalize syntax and substitute local hex addresses with symbolic labels
    out_lines = []
    replaced_count = 0

    for line in raw_lines:
        stripped = line.strip()
        if not stripped or stripped.startswith(";") or stripped.startswith("org"):
            out_lines.append(line)
            continue

        label_part = ""
        inst_part = line
        if ":" in line and not line.lstrip().startswith(";"):
            idx = line.index(":")
            label_part = line[:idx + 1]
            inst_part = line[idx + 1:]

        inst_part = normalize_syntax(inst_part)

        op_match = re.match(r"^\\s*([a-z]+)(?:\\.[bws])?\\s+(.+)$", inst_part, re.IGNORECASE)
        if op_match:
            mnemonic = op_match.group(1).lower()
            operands = op_match.group(2)

            if mnemonic in BRANCH_OR_PTR_MNEMONICS:
                def sub_addr(match):
                    nonlocal replaced_count
                    raw_hex = match.group(1)
                    suffix = match.group(2) or ""

                    if target_int in symbol_map:

                        return f"{symbol_map[target_int]}{suffix}"
                    return match.group(0)

                    r"(?<!#)\\$([0-9a-fA-F]{4,8})(\\(pc\\))?",
                    sub_addr,
                    operands,
                    flags=re.IGNORECASE
                )
                inst_part = re.sub(
                    r"^(\\s*[a-z]+(?:\\.[bws])?\\s+).+$",
                    rf"\\g<1>{operands}",
                    inst_part,
                    flags=re.IGNORECASE
                )

        out_lines.append(f"{label_part}{inst_part}")

    with open(path, "w") as f:
        f.writelines(out_lines)

    print(f"[SYMBOLIC OK] {os.path.basename(path)}: {len(symbol_map)} labels indexed, {replaced_count} operands resolved")

files["genesis_subroutines.asm"] = """; =============================================================================
; STANDALONE SEGA GENESIS (MOTOROLA 68000) HARDWARE SUBROUTINES
; Assembler Target: vasmm68k_mot -Fbin -m68000
; =============================================================================

; -----------------------------------------------------------------------------
; 1. VDP CONTROL REGISTER INITIALIZATION
; -----------------------------------------------------------------------------
Initialize_VDP:
    lea     VDP_Register_Data, a0   ; Point a0 to configuration data array
    move.w  #23, d0                 ; Loop counter: 24 registers total (0 to 23)
    move.w  #$8000, d1              ; Command register base bit parameter

.VDP_Load_Loop:
    move.b  (a0)+, d1               ; Fetch byte value into lower d1
    move.w  d1, $00C00004           ; Write word command to VDP Control Port
    add.w   #$0100, d1              ; Increment register index in upper byte
    dbra    d0, .VDP_Load_Loop
    rts

VDP_Register_Data:
    dc.b $04, $14, $30, $3C, $07, $6C, $00, $00
    dc.b $00, $00, $FF, $00, $81, $37, $00, $02
    dc.b $01, $00, $00, $00, $00, $00, $00, $00
    even

; -----------------------------------------------------------------------------
; 2. CLEAR ALL 64KB ($0000-$FFFF) OF VIDEO RAM
; -----------------------------------------------------------------------------
Clear_VRAM:
    move.l  #$8F02C000, $00C00004   ; Set VDP Auto-Increment = 2 bytes
    move.l  #$40000000, $00C00004   ; Set VDP write destination to VRAM $0000
    move.w  #$7FFF, d0              ; 32,768 words (64KB)
    moveq   #0, d1

.VRAM_Clear_Loop:
    move.w  d1, $00C00000
    dbra    d0, .VRAM_Clear_Loop
    rts

; -----------------------------------------------------------------------------
; 3. PRINT NULL-TERMINATED ASCII STRING TO VRAM NAMETABLE
; Inputs: a0 = pointer to null-terminated string, d0.l = VRAM offset command bits
; -----------------------------------------------------------------------------
Print_ASCII_String:
    move.l  #$40000000, d1
    or.l    d0, d1
    move.l  d1, $00C00004

.Print_Character_Loop:
    move.b  (a0)+, d2
    beq.s   .Print_Done
    ext.w   d2
    andi.w  #$00FF, d2
    sub.w   #$0020, d2
    add.w   #$0001, d2
