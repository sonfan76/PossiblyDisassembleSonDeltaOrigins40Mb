#!/usr/bin/env python3
"""
match_upstream.py
Matches binary assets, palettes, art, sound data, and compiled code blocks
from s1disasm-AS, s2disasm-master, and skdisasm-master against the 40Mb ROM.
"""

import os
import sys
import json
import hashlib
import argparse
from pathlib import Path

BANK_SIZE = 0x80000  # 512 KB per SSF2 bank
MIN_MATCH_SIZE = 64  # Ignore tiny files (< 64 bytes) to avoid false positives

ASSET_EXTENSIONS = {
    ".bin", ".nem", ".kos", ".kosm", ".eni", ".unc", ".wav"
}


def index_reference_repos(repo_dirs):
    """Build a lookup table of SHA-256 hashes and byte signatures from upstream repos."""
    exact_hashes = {}
    prefix_index = {}

    for label, root_path in repo_dirs.items():
        root = Path(root_path)
        if not root.exists():
            print(f"[!] Warning: Reference directory not found: {root}")
            continue

        count = 0
        for filepath in root.rglob("*"):
            if not filepath.is_file():
                continue
            if filepath.suffix.lower() not in ASSET_EXTENSIONS:
                continue

            data = filepath.read_bytes()
            size = len(data)
            if size < MIN_MATCH_SIZE:
                continue

            rel_path = str(filepath.relative_to(root))
            digest = hashlib.sha256(data).hexdigest()

            entry = {
                "repo": label,
                "path": rel_path,
                "size": size,
                "sha256": digest,
            }
            exact_hashes.setdefault(digest, []).append(entry)

            # Index by first 32 bytes for fast linear scanning across the 5MB ROM
            prefix = data[:32]
            prefix_index.setdefault(prefix, []).append((data, entry))
            count += 1

        print(f"[+] Indexed {count} binary assets from {label} ({root})")

    return exact_hashes, prefix_index


def scan_rom(rom_path, prefix_index):
    """Scan the 40Mbit ROM for verbatim upstream assets and data tables."""
    rom_data = Path(rom_path).read_bytes()
    rom_size = len(rom_data)
    print(f"[*] Scanning {rom_path} ({rom_size:,} bytes / {rom_size // BANK_SIZE} banks)...")

    matches = []
    pos = 0
    step = 2  # M68K word-aligned boundary

    while pos <= rom_size - 32:
        prefix = rom_data[pos : pos + 32]
        candidates = prefix_index.get(prefix)
        if candidates:
            matched_here = False
            for ref_bytes, meta in candidates:
                end_pos = pos + meta["size"]
                if end_pos <= rom_size and rom_data[pos:end_pos] == ref_bytes:
                    bank_id = pos // BANK_SIZE
                    bank_offset = pos % BANK_SIZE
                    matches.append({
                        "rom_offset": f"0x{pos:06X}",
                        "rom_offset_int": pos,
                        "bank": bank_id,
                        "bank_offset": f"0x{bank_offset:05X}",
                        "size": meta["size"],
                        "repo": meta["repo"],
                        "upstream_path": meta["path"],
                    })
                    pos = (end_pos + 1) & ~1  # Advance past matched block (word-aligned)
                    matched_here = True
                    break
            if matched_here:
                continue
        pos += step

    return matches


def main():
    parser = argparse.ArgumentParser(description="Match upstream Sonic disassemblies against 40Mb ROM")
    parser.add_argument("--rom", default="SONIC_DELTA_40Mb.bin", help="Path to 40Mbit ROM")
    parser.add_argument("--s1", default="s1disasm-AS", help="Path to s1disasm-AS directory")
    parser.add_argument("--s2", default="s2disasm-master", help="Path to s2disasm-master directory")
    parser.add_argument("--sk", default="skdisasm-master", help="Path to skdisasm-master directory")
    parser.add_argument("--out", default="upstream_matches.json", help="Output JSON report")
    args = parser.parse_args()

    repos = {
        "s1disasm": args.s1,
        "s2disasm": args.s2,
        "skdisasm": args.sk,
    }

    _, prefix_index = index_reference_repos(repos)
    matches = scan_rom(args.rom, prefix_index)

    # Summarize coverage per bank
    bank_summary = {}
    for m in matches:
        b = m["bank"]
        bank_summary[b] = bank_summary.get(b, 0) + m["size"]

    report = {
        "rom": args.rom,
        "total_matches": len(matches),
        "matched_bytes_per_bank": {f"bank_{k:02d}": v for k, v in sorted(bank_summary.items())},
        "matches": matches,
    }

    Path(args.out).write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(f"[+] Found {len(matches)} verbatim upstream blocks -> saved to {args.out}")
    for b, total in sorted(bank_summary.items()):
        pct = (total / BANK_SIZE) * 100
        print(f"    Bank {b:02d}: {total:,} / {BANK_SIZE:,} bytes matched ({pct:.1f}%)")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
split_to_repo.py
Splits the 40Mbit ROM and disassembled bank ASM files into a clean,
modular repository structure modeled on skdisasm / s2disasm.
"""

import os
import json
import shutil
import argparse
from pathlib import Path

BANK_SIZE = 0x80000  # 512 KB per SSF2 bank


def sanitize_label(path_str):
    """Convert an upstream relative path into a valid M68K assembly label."""
    stem = Path(path_str).stem
    clean = "".join(c if c.isalnum() else "_" for c in stem)
    while "__" in clean:
        clean = clean.replace("__", "_")
    return clean.strip("_")


def build_repo_structure(rom_path, matches_json, asm_dir, out_repo):
    rom_data = Path(rom_path).read_bytes()
    report = json.loads(Path(matches_json).read_text(encoding="utf-8"))
    matches = report.get("matches", [])

    out = Path(out_repo)
    dirs = {
        "src": out / "src",
        "banks": out / "src" / "banks",
        "custom": out / "src" / "custom_neto",
        "assets": out / "assets",
        "tools": out / "tools",
    }
    for d in dirs.values():
        d.mkdir(parents=True, exist_ok=True)

    # 1. Extract matched upstream binary assets into organized folders
    asset_inc_lines = [
        "; ============================================================================",
        "; Auto-generated Upstream Asset Table (s1disasm / s2disasm / skdisasm)",
        "; ============================================================================",
        "",
    ]

    for m in matches:
        start = m["rom_offset_int"]
        size = m["size"]
        repo_tag = m["repo"]
        rel_upstream = Path(m["upstream_path"])

        dest_file = dirs["assets"] / repo_tag / rel_upstream
        dest_file.parent.mkdir(parents=True, exist_ok=True)
        dest_file.write_bytes(rom_data[start : start + size])

        lbl = f"{repo_tag}_{sanitize_label(str(rel_upstream))}_{start:06X}"
        rel_inc = dest_file.relative_to(out).as_posix()
        asset_inc_lines.append(f"; ROM Offset: 0x{start:06X} (Bank {m['bank']:02d} @ {m['bank_offset']})")
        asset_inc_lines.append(f"{lbl}:")
        asset_inc_lines.append(f'    incbin "{rel_inc}"')
        asset_inc_lines.append("")

    (dirs["src"] / "upstream_assets.asm").write_text("\n".join(asset_inc_lines), encoding="utf-8")

    # 2. Copy and organize disassembled M68K banks
    asm_source = Path(asm_dir)
    bank_files = sorted(asm_source.glob("bank_*.asm")) if asm_source.exists() else []

    master_lines = [
        "; ============================================================================",
        "; Sonic Delta 40Mb (v0.79) - Reconstructed Open-Source M68K Disassembly",
        "; Target Assembler: vasm (vasmm68k_mot -Fbin -spaces)",
        "; ============================================================================",
        "",
        '    include "src/ssf2_mapper.inc"',
        '    include "src/upstream_assets.asm"',
        "",
    ]

    for bf in bank_files:
        dest_bf = dirs["banks"] / bf.name
        shutil.copy2(bf, dest_bf)
        master_lines.append(f'    include "src/banks/{bf.name}"')

    (out / "main.asm").write_text("\n".join(master_lines), encoding="utf-8")

    # 3. Write SSF2 Hardware Mapper & SRAM macro definitions (documented from ROM Exceção $41)
    mapper_inc = """\
; ============================================================================
; Sega SSF2 Bank Mapper & SRAM Control Registers ($A130F1 - $A130FF)
; Verified from ROM Exception Handler (Excecao $41: SRAM nao disponivel)
; ============================================================================

SRAM_CTRL       equ $A130F1   ; Write 1 to enable SRAM at $200000-$3FFFFF, 0 for ROM
BANK_REG_1      equ $A130F3   ; Maps $080000 - $0FFFFF (512 KB)
BANK_REG_2      equ $A130F5   ; Maps $100000 - $17FFFF (512 KB)
BANK_REG_3      equ $A130F7   ; Maps $180000 - $1FFFFF (512 KB)
BANK_REG_4      equ $A130F9   ; Maps $200000 - $27FFFF (512 KB)
BANK_REG_5      equ $A130FB   ; Maps $280000 - $2FFFFF (512 KB)
BANK_REG_6      equ $A130FD   ; Maps $300000 - $37FFFF (512 KB)
BANK_REG_7      equ $A130FF   ; Maps $380000 - $3FFFFF (512 KB)

SET_BANK macro reg, bank_id
    move.b  #\\bank_id, (\\reg).l
    endm
"""
    (dirs["src"] / "ssf2_mapper.inc").write_text(mapper_inc, encoding="utf-8")

    # 4. Write repository Makefile for vasm rebuild & SHA-256 verification
    makefile_text = f"""\
VASM = vasmm68k_mot
VASMFLAGS = -Fbin -spaces -devpac
TARGET = build/sonic_delta_40mb_rebuilt.bin
ORIG_ROM = {rom_path}

all: $(TARGET) verify

$(TARGET): main.asm
\t@mkdir -p build
\t$(VASM) $(VASMFLAGS) -o $(TARGET) main.asm

verify: $(TARGET)
\t@echo "Comparing rebuilt binary against original ROM..."
\t@cmp -l $(ORIG_ROM) $(TARGET) | head -n 20 || true
\t@sha256sum $(ORIG_ROM) $(TARGET)

clean:
\trm -rf build
"""
    (out / "Makefile").write_text(makefile_text, encoding="utf-8")
    print(f"[+] Repository scaffold generated at: {out.resolve()}")
    print(f"[+] Extracted {len(matches)} upstream assets and linked {len(bank_files)} bank files.")


def main():
    parser = argparse.ArgumentParser(description="Build modular Git repo from 40Mb ROM & matches")
    parser.add_argument("--rom", default="SONIC_DELTA_40Mb.bin", help="Path to 40Mbit ROM")
    parser.add_argument("--matches", default="upstream_matches.json", help="Path to upstream_matches.json")

#!/usr/bin/env python3
"""
match_upstream.py (v2)
Matches both binary assets AND compiled 68000 code routines from
s1disasm-AS, s2disasm-master, and skdisasm-master against the 40Mb ROM.
Supports opcode-masked signature matching so routines relocated to new
SSF2 banks still match their upstream labels.
"""

import re
import json
import struct
import hashlib
import argparse
from pathlib import Path

BANK_SIZE = 0x80000      # 512 KB per SSF2 bank
MIN_ASSET_SIZE = 64      # Minimum bytes for binary asset matching
MIN_CODE_SIZE = 24       # Minimum bytes for 68K routine matching

ASSET_EXTENSIONS = {".bin", ".nem", ".kos", ".kosm", ".eni", ".unc", ".wav"}


def mask_m68k_relocations(code_bytes):
    """
    Create a normalized signature by zeroing out 32-bit absolute address operands
    in common M68K instructions (JSR/JMP/LEA/MOVE.L #imm, An) so relocated code matches.
    """
    buf = bytearray(code_bytes)
    n = len(buf)
    i = 0
    while i + 1 < n:
        op = (buf[i] << 8) | buf[i + 1]
        # JSR (xxx).L ($4EB9) or JMP (xxx).L ($4EF9) -> mask next 4 bytes
        if op in (0x4EB9, 0x4EF9) and i + 6 <= n:
            buf[i + 2 : i + 6] = b"\x00\x00\x00\x00"
            i += 6
            continue
        # LEA (xxx).L, An ((op & 0xF1FF) == 0x41F9) -> mask next 4 bytes
        if (op & 0xF1FF) == 0x41F9 and i + 6 <= n:
            buf[i + 2 : i + 6] = b"\x00\x00\x00\x00"
            i += 6
            continue
        # MOVEA.L #imm, An ((op & 0xF1FF) == 0x207C) -> mask next 4 bytes
        if (op & 0xF1FF) == 0x207C and i + 6 <= n:
            buf[i + 2 : i + 6] = b"\x00\x00\x00\x00"
            i += 6
            continue
        i += 2
    return bytes(buf)


def parse_as_listing(lst_path):
    """
    Parse an AS assembler .lst file to extract (label_name, rom_offset, byte_length).
    """
    symbols = []
    # Matches AS listing lines like: "  1234/    1A2B : 4E75          MyLabel:"
    label_re = re.compile(r"^\s*\d+/\s*([0-9A-Fa-f]+)\s*:\s*([A-Za-z_][A-Za-z0-9_]*)\s*:")
    text = lst_path.read_text(encoding="utf-8", errors="ignore")
    for line in text.splitlines():
        m = label_re.match(line)
        if m:
            addr = int(m.group(1), 16)
            name = m.group(2)
            # Filter out local/anonymous AS labels
            if not name.startswith(("loc_", ".", "@", "+")):
                symbols.append((name, addr))

    # Deduplicate and sort by address
    symbols = sorted(set(symbols), key=lambda x: x[1])
    routines = []
    for idx, (name, start) in enumerate(symbols):
        end = symbols[idx + 1][1] if idx + 1 < len(symbols) else start
        size = end - start
        if MIN_CODE_SIZE <= size <= 0x4000:
            routines.append((name, start, size))
    return routines


def index_reference_repos(repo_dirs):
    """Index binary assets and (if built ROM + .lst exist) 68000 code routines."""
    asset_prefix_index = {}
    code_exact_index = {}
    code_masked_index = {}

    for label, root_path in repo_dirs.items():
        root = Path(root_path)
        if not root.exists():
            print(f"[!] Warning: Reference directory not found: {root}")
            continue

        asset_count = 0
        for filepath in root.rglob("*"):
            if not filepath.is_file():
                continue
            if filepath.suffix.lower() not in ASSET_EXTENSIONS:
                continue

            data = filepath.read_bytes()
            if len(data) < MIN_ASSET_SIZE:
                continue

            rel_path = str(filepath.relative_to(root))
            entry = {
                "type": "asset",
                "repo": label,
                "name": rel_path,
                "size": len(data),
            }
            asset_prefix_index.setdefault(data[:32], []).append((data, entry))
            asset_count += 1

        print(f"[+] Indexed {asset_count} binary assets from {label}")

        # Look for built reference ROMs (.bin) paired with AS listings (.lst)
        code_count = 0
        for lst_file in root.glob("*.lst"):
            bin_candidate = lst_file.with_suffix(".bin")
            if not bin_candidate.exists():
                continue
            ref_rom = bin_candidate.read_bytes()
            routines = parse_as_listing(lst_file)
            for sym_name, start, size in routines:
                if start + size > len(ref_rom):
                    continue
                chunk = ref_rom[start : start + size]
                masked = mask_m68k_relocations(chunk)
                meta = {
                    "type": "code",
                    "repo": label,
                    "name": sym_name,
                    "ref_offset": f"0x{start:06X}",
                    "size": size,
                }
                code_exact_index.setdefault(chunk[:16], []).append((chunk, meta))
                code_masked_index.setdefault(masked[:16], []).append((masked, meta))
                code_count += 1

        if code_count > 0:
            print(f"[+] Indexed {code_count} compiled 68K routines from {label} ({lst_file.name})")
        else:
            print(f"[i] No built .bin/.lst pair found in {label} (run build.bat/lua to enable code matching)")

    return asset_prefix_index, code_exact_index, code_masked_index


def scan_rom(rom_path, asset_idx, code_exact_idx, code_masked_idx):
    """Scan the 40Mb ROM for verbatim assets, exact 68K code, and relocated 68K code."""
    rom_data = Path(rom_path).read_bytes()
    rom_size = len(rom_data)
    print(f"[*] Scanning {rom_path} ({rom_size:,} bytes / {rom_size // BANK_SIZE} banks)...")

    matches = []
    pos = 0

    while pos <= rom_size - MIN_CODE_SIZE:
        matched = False

        # 1. Check verbatim binary assets (32-byte prefix)
        if pos <= rom_size - MIN_ASSET_SIZE:
            p32 = rom_data[pos : pos + 32]
            if p32 in asset_idx:
                for ref_bytes, meta in asset_idx[p32]:
                    end_pos = pos + meta["size"]
                    if end_pos <= rom_size and rom_data[pos:end_pos] == ref_bytes:
                        matches.append({
                            "rom_offset": f"0x{pos:06X}",
                            "rom_offset_int": pos,
                            "bank": pos // BANK_SIZE,
                            "bank_offset": f"0x{pos % BANK_SIZE:05X}",
                            "size": meta["size"],
                            "kind": "asset",
                            "match_mode": "exact",
                            "repo": meta["repo"],
                            "symbol": meta["name"],
                        })
                        pos = (end_pos + 1) & ~1
                        matched = True
                        break
        if matched:
            continue

        # 2. Check exact 68000 machine code (16-byte prefix)
        p16 = rom_data[pos : pos + 16]
        if p16 in code_exact_idx:
            for ref_bytes, meta in code_exact_idx[p16]:
                end_pos = pos + meta["size"]
                if end_pos <= rom_size and rom_data[pos:end_pos] == ref_bytes:
                    matches.append({
                        "rom_offset": f"0x{pos:06X}",
                        "rom_offset_int": pos,
                        "bank": pos // BANK_SIZE,
                        "bank_offset": f"0x{pos % BANK_SIZE:05X}",
                        "size": meta["size"],
                        "kind": "code",
                        "match_mode": "exact",
                        "repo": meta["repo"],
                        "symbol": meta["name"],
                    })
                    pos = (end_pos + 1) & ~1
                    matched = True
                    break
        if matched:
            continue

        # 3. Check relocated 68000 machine code (address-masked signature)
        if code_masked_idx:
            window_masked = mask_m68k_relocations(rom_data[pos : pos + 16])
            if window_masked in code_masked_idx:
                for ref_masked, meta in code_masked_idx[window_masked]:
                    end_pos = pos + meta["size"]
                    if end_pos <= rom_size:
                        cand_masked = mask_m68k_relocations(rom_data[pos:end_pos])
                        if cand_masked == ref_masked:
                            matches.append({
                                "rom_offset": f"0x{pos:06X}",
                                "rom_offset_int": pos,
                                "bank": pos // BANK_SIZE,
                                "bank_offset": f"0x{pos % BANK_SIZE:05X}",
                                "size": meta["size"],
                                "kind": "code",
                                "match_mode": "relocated",
                                "repo": meta["repo"],
                                "symbol": meta["name"],
                            })
                            pos = (end_pos + 1) & ~1
                            matched = True
                            break
        if matched:
            continue

        pos += 2  # Advance by 1 M68K word

    return matches

### 2. `ghidra_auto_bank.py` (v2 — Upstream Symbol & Asset Importer)

Drop this updated script into your toolkit to replace the v1 `ghidra_auto_bank.py`. When Ghidra runs on your 10 bank binaries (`bank_00.bin`–`bank_09.bin`), it automatically:
- Maps the Sega Genesis hardware I/O registers (`$A130F1`–`$A130FF` SSF2 mapper, `$C00000`–`$C00004` VDP, `$A10003`–`$A1000D` Joypads) so Ghidra labels them cleanly.
- Reads `upstream_matches.json`, creates primary symbols for every matched routine/asset in that bank, and **converts matched binary art/sound ranges into byte data** (`clearListing` + `createData`) so Ghidra's auto-analyzer doesn't corrupt your disassembly by trying to decode compressed graphics as 68000 instructions.

```python
# ghidra_auto_bank.py (v2)
# Run headless via Ghidra analyzeHeadless:
# analyzeHeadless ./ghidra_proj SonicDelta -import bank_00.bin -processor 68000:BE:32:default -postScript ghidra_auto_bank.py

import os
import json
from ghidra.program.model.symbol import SourceType
from ghidra.program.model.data import ByteDataType, ArrayDataType
from ghidra.app.cmd.disassemble import DisassembleCommand

BANK_SIZE = 0x80000


def sanitize_symbol(name):
    clean = "".join(c if c.isalnum() else "_" for c in name)
    while "__" in clean:
        clean = clean.replace("__", "_")
    return clean.strip("_")


def label_genesis_hardware():
    """Label standard Mega Drive / Genesis and Sega SSF2 mapper registers."""
    hw_regs = {
        0x00A10001: "HW_VERSION",
        0x00A10003: "JOYPAD_1_DATA",
        0x00A10005: "JOYPAD_2_DATA",
        0x00A10009: "JOYPAD_1_CTRL",
        0x00A1000B: "JOYPAD_2_CTRL",
        0x00A11100: "Z80_BUSREQ",
        0x00A11200: "Z80_RESET",
        0x00A130F1: "SSF2_SRAM_CTRL",
        0x00A130F3: "SSF2_BANK_1",
        0x00A130F5: "SSF2_BANK_2",
        0x00A130F7: "SSF2_BANK_3",
        0x00A130F9: "SSF2_BANK_4",
        0x00A130FB: "SSF2_BANK_5",
        0x00A130FD: "SSF2_BANK_6",
        0x00A130FF: "SSF2_BANK_7",
        0x00C00000: "VDP_DATA",
        0x00C00004: "VDP_CTRL",
        0x00C00008: "VDP_HV_COUNTER",
        0x00C00011: "PSG_CTRL",
    }
    sym_table = currentProgram.getSymbolTable()
    addr_factory = currentProgram.getAddressFactory()
    space = addr_factory.getDefaultAddressSpace()

    for addr_int, name in hw_regs.items():
        addr = space.getAddress(addr_int)
        try:
            sym_table.createLabel(addr, name, SourceType.USER_DEFINED)
        except Exception:
            pass


def apply_upstream_matches(bank_index):
    """Import symbols and block off binary assets from upstream_matches.json."""
    matches_path = "upstream_matches.json"
    if not os.path.exists(matches_path):
        print("[!] upstream_matches.json not found; skipping upstream symbol import.")
        return

    with open(matches_path, "r") as f:
        report = json.load(f)

    sym_table = currentProgram.getSymbolTable()
    listing = currentProgram.getListing()
    space = currentProgram.getAddressFactory().getDefaultAddressSpace()
    base_addr = currentProgram.getImageBase().getOffset()

    applied = 0
    for m in report.get("matches", []):
        if m.get("bank") != bank_index:
            continue

        local_offset = int(m["bank_offset"], 16)
        size = m["size"]
        kind = m["kind"]
        repo = m["repo"]
        sym_raw = m["symbol"]
        mode = m.get("match_mode", "exact")

        target_addr = space.getAddress(base_addr + local_offset)
        sym_name = "{}_{}_{:05X}".format(repo, sanitize_symbol(sym_raw), local_offset)

        try:
            sym_table.createLabel(target_addr, sym_name, SourceType.USER_DEFINED)
            comment = "[UPSTREAM: {} | {} | {} | {} bytes]".format(repo, kind, mode, size)
            listing.setComment(target_addr, 0, comment)  # EOL_COMMENT = 0
        except Exception as e:
            print("[!] Label error at 0x{:05X}: {}".format(local_offset, e))

        # If this match is a binary asset (Nemesis/Kosinski/PCM/etc.), prevent code disassembly
        if kind == "asset":
            end_addr = space.getAddress(base_addr + local_offset + size - 1)
            listing.clearCodeUnits(target_addr, end_addr, False)
            try:
                arr_type = ArrayDataType(ByteDataType.dataType, size, 1)
                listing.createData(target_addr, arr_type)
            except Exception:
                pass
        elif kind == "code":
            cmd = DisassembleCommand(target_addr, None, True)
            cmd.applyTo(currentProgram)

        applied += 1

    print("[+] Applied {} upstream symbols/assets to Bank {:02d}".format(applied, bank_index))


def run():
    prog_name = currentProgram.getName()
    bank_index = 0
    # Infer bank index from filename (e.g., bank_03.bin -> 3)
    for part in prog_name.replace(".", "_").split("_"):
        if part.isdigit():
            bank_index = int(part)
            break

    print("[*] Running ghidra_auto_bank v2 on {} (Bank {:02d})".format(prog_name, bank_index))
    label_genesis_hardware()
    apply_upstream_matches(bank_index)


run()
