#!/usr/bin/env python3
"""
match_upstream.py (v2)
Matches both binary assets AND compiled 68000 code routines from
s1disasm-AS, s2disasm-master, and skdisasm-master against the 40Mb ROM.
Supports opcode-masked signature matching so routines relocated to new
SSF2 banks still match their upstream labels.
"""

import re
import json
import argparse
from pathlib import Path

BANK_SIZE = 0x80000      # 512 KB per SSF2 bank
MIN_ASSET_SIZE = 64      # Minimum bytes for binary asset matching
MIN_CODE_SIZE = 24       # Minimum bytes for 68K routine matching
ASSET_EXTENSIONS = {".bin", ".nem", ".kos", ".kosm", ".eni", ".unc", ".wav"}


def mask_m68k_relocations(code_bytes):
    """
    Create a normalized signature by zeroing out 32-bit absolute address operands
    in common M68K instructions (JSR/JMP/LEA/MOVEA.L #imm, An) so relocated code matches.
    """
    buf = bytearray(code_bytes)
    n = len(buf)
    i = 0
    while i + 1 < n:
        op = (buf[i] << 8) | buf[i + 1]
        # JSR (xxx).L ($4EB9) or JMP (xxx).L ($4EF9) -> mask next 4 bytes
        if op in (0x4EB9, 0x4EF9) and i + 6 <= n:
            buf[i + 2 : i + 6] = b"\x00\x00\x00\x00"
            i += 6
            continue
        # LEA (xxx).L, An ((op & 0xF1FF) == 0x41F9) -> mask next 4 bytes
        if (op & 0xF1FF) == 0x41F9 and i + 6 <= n:
            buf[i + 2 : i + 6] = b"\x00\x00\x00\x00"
            i += 6
            continue
        # MOVEA.L #imm, An ((op & 0xF1FF) == 0x207C) -> mask next 4 bytes
        if (op & 0xF1FF) == 0x207C and i + 6 <= n:
            buf[i + 2 : i + 6] = b"\x00\x00\x00\x00"
            i += 6
            continue
        i += 2
    return bytes(buf)


def parse_as_listing(lst_path):
    """
    Parse an AS assembler .lst file to extract (label_name, rom_offset, byte_length).
    """
    symbols = []
    label_re = re.compile(r"^\s*\d+/\s*([0-9A-Fa-f]+)\s*:\s*([A-Za-z_][A-Za-z0-9_]*)\s*:")
    text = lst_path.read_text(encoding="utf-8", errors="ignore")
    for line in text.splitlines():
        m = label_re.match(line)
        if m:
            addr = int(m.group(1), 16)
            name = m.group(2)
            if not name.startswith(("loc_", ".", "@", "+")):
                symbols.append((name, addr))

    symbols = sorted(set(symbols), key=lambda x: x[1])
    routines = []
    for idx, (name, start) in enumerate(symbols):
        end = symbols[idx + 1][1] if idx + 1 < len(symbols) else start
        size = end - start
        if MIN_CODE_SIZE <= size <= 0x4000:
            routines.append((name, start, size))
    return routines


def index_reference_repos(repo_dirs):
    """Index binary assets and (if built ROM + .lst exist) 68000 code routines."""
    asset_prefix_index = {}
    code_exact_index = {}
    code_masked_index = {}

    for label, root_path in repo_dirs.items():
        root = Path(root_path)
        if not root.exists():
            print(f"[!] Warning: Reference directory not found: {root}")
            continue

        asset_count = 0
        for filepath in root.rglob("*"):
            if not filepath.is_file():
                continue
            if filepath.suffix.lower() not in ASSET_EXTENSIONS:
                continue
            data = filepath.read_bytes()
            if len(data) < MIN_ASSET_SIZE:
                continue

            rel_path = str(filepath.relative_to(root))
            entry = {
                "type": "asset",
                "repo": label,
                "name": rel_path,
                "size": len(data),
            }
            asset_prefix_index.setdefault(data[:32], []).append((data, entry))
            asset_count += 1
        print(f"[+] Indexed {asset_count} binary assets from {label}")

        code_count = 0
        for lst_file in root.glob("*.lst"):
            bin_candidate = lst_file.with_suffix(".bin")
            if not bin_candidate.exists():
                continue
            ref_rom = bin_candidate.read_bytes()
            routines = parse_as_listing(lst_file)
            for sym_name, start, size in routines:
                if start + size > len(ref_rom):
                    continue
                chunk = ref_rom[start : start + size]
                masked = mask_m68k_relocations(chunk)
                meta = {
                    "type": "code",
                    "repo": label,
                    "name": sym_name,
                    "ref_offset": f"0x{start:06X}",
                    "size": size,
                }
                code_exact_index.setdefault(chunk[:16], []).append((chunk, meta))
                code_masked_index.setdefault(masked[:16], []).append((masked, meta))
                code_count += 1
        if code_count > 0:
            print(f"[+] Indexed {code_count} compiled 68K routines from {label}")
        else:
            print(f"[i] No built .bin/.lst pair found in {label} (run build.bat/lua to enable code matching)")

    return asset_prefix_index, code_exact_index, code_masked_index


def scan_rom(rom_path, asset_idx, code_exact_idx, code_masked_idx):
    """Scan the 40Mb ROM for verbatim assets, exact 68K code, and relocated 68K code."""
    rom_data = Path(rom_path).read_bytes()
    rom_size = len(rom_data)
    print(f"[*] Scanning {rom_path} ({rom_size:,} bytes / {rom_size // BANK_SIZE} banks)...")

    matches = []
    pos = 0
    while pos <= rom_size - MIN_CODE_SIZE:
        matched = False

        # 1. Check verbatim binary assets (32-byte prefix)
        if pos <= rom_size - MIN_ASSET_SIZE:
            p32 = rom_data[pos : pos + 32]
            if p32 in asset_idx:
                for ref_bytes, meta in asset_idx[p32]:
                    end_pos = pos + meta["size"]
                    if end_pos <= rom_size and rom_data[pos:end_pos] == ref_bytes:
                        matches.append({
                            "rom_offset": f"0x{pos:06X}",
                            "rom_offset_int": pos,
                            "bank": pos // BANK_SIZE,
                            "bank_offset": f"0x{pos % BANK_SIZE:05X}",
                            "size": meta["size"],
                            "kind": "asset",
                            "match_mode": "exact",
                            "repo": meta["repo"],
                            "symbol": meta["name"],
                        })
                        pos = (end_pos + 1) & ~1
                        matched = True
                        break
        if matched:
            continue

        # 2. Check exact 68000 machine code (16-byte prefix)
        p16 = rom_data[pos : pos + 16]
        if p16 in code_exact_idx:
            for ref_bytes, meta in code_exact_idx[p16]:
                end_pos = pos + meta["size"]
                if end_pos <= rom_size and rom_data[pos:end_pos] == ref_bytes:
                    matches.append({
                        "rom_offset": f"0x{pos:06X}",
                        "rom_offset_int": pos,
                        "bank": pos // BANK_SIZE,
                        "bank_offset": f"0x{pos % BANK_SIZE:05X}",
                        "size": meta["size"],
                        "kind": "code",
                        "match_mode": "exact",
                        "repo": meta["repo"],
                        "symbol": meta["name"],
                    })
                    pos = (end_pos + 1) & ~1
                    matched = True
                    break
        if matched:
            continue

        # 3. Check relocated 68000 machine code (address-masked signature)
        if code_masked_idx:
            window_masked = mask_m68k_relocations(rom_data[pos : pos + 16])
            if window_masked in code_masked_idx:
                for ref_masked, meta in code_masked_idx[window_masked]:
                    end_pos = pos + meta["size"]
                    if end_pos <= rom_size:
                        cand_masked = mask_m68k_relocations(rom_data[pos:end_pos])
                        if cand_masked == ref_masked:
                            matches.append({
                                "rom_offset": f"0x{pos:06X}",
                                "rom_offset_int": pos,
                                "bank": pos // BANK_SIZE,
                                "bank_offset": f"0x{pos % BANK_SIZE:05X}",
                                "size": meta["size"],
                                "kind": "code",
                                "match_mode": "relocated",
                                "repo": meta["repo"],
                                "symbol": meta["name"],
                            })
                            pos = (end_pos + 1) & ~1
                            matched = True
                            break
        if matched:
            continue

        pos += 2  # Advance by 1 M68K word

    return matches


def main():
    parser = argparse.ArgumentParser(description="Match upstream assets & 68K code against 40Mb ROM")
    parser.add_argument("--rom", default="Sonic_Delta_Origins_2026.bin", help="Path to 40Mbit ROM")
    parser.add_argument("--s1", default="s1disasm-AS", help="Path to s1disasm-AS")

        b = m["bank"]

    report = {
        "rom": args.rom,
        "total_matches": len(matches),

Use code with caution.

Path(args.out).write_text(json.dumps(report, indent=2), encoding="utf-8")

#!/usr/bin/env python3
"""
extract_neto_custom.py
Scans Sonic_Delta_Origins_2026.bin (40Mbit SSF2 ROM) for Neto's custom code:
  1. All 68000 instructions reading/writing SSF2 mapper registers ($A130F1-$A130FF)
  2. Embedded Portuguese debug/exception strings & string pointer references
  3. Bank header signatures ("M68K Bank X") and NETO padding blocks
Outputs neto_custom_hooks.json and src/custom_neto/neto_hooks_map.inc.
"""

import json
import struct
import argparse
from pathlib import Path

BANK_SIZE = 0x80000  # 512 KB per SSF2 bank

SSF2_REGS = {
    0x00A130F1: "SSF2_SRAM_CTRL",
    0x00A130F3: "SSF2_BANK_1",
    0x00A130F5: "SSF2_BANK_2",
    0x00A130F7: "SSF2_BANK_3",
    0x00A130F9: "SSF2_BANK_4",
    0x00A130FB: "SSF2_BANK_5",
    0x00A130FD: "SSF2_BANK_6",
    0x00A130FF: "SSF2_BANK_7",
}

NETO_SIGNATURES = [
    b"Neto - Depurador",
    b"Super Depurador",
    b"Mapeador n",
    b"SRAM n",
    b"Sonic Delta Next",
    b"SONIC DELTA 40Mb",
    b"M68K Bank",
    b"neto.baiano",
]


def decode_simple_m68k(rom, offset):
    """
    Lightweight decoder for common M68K instructions around SSF2 mapper accesses.
    """
    if offset + 2 > len(rom):
        return "dc.w $0000", 2

    op = struct.unpack(">H", rom[offset : offset + 2])[0]

    # MOVE.B #imm, (xxx).L -> opcode 0x13FC, imm word (low byte), 32-bit addr
    if op == 0x13FC and offset + 8 <= len(rom):
        imm = struct.unpack(">H", rom[offset + 2 : offset + 4])[0] & 0xFF
        addr = struct.unpack(">I", rom[offset + 4 : offset + 8])[0]
        reg_name = SSF2_REGS.get(addr, f"${addr:08X}")
        return f"move.b  #${imm:02X}, ({reg_name}).l", 8

    # MOVE.B Dn, (xxx).L -> (op & 0xFFF8) == 0x13C0, 32-bit addr
    if (op & 0xFFF8) == 0x13C0 and offset + 6 <= len(rom):
        dn = op & 0x0007
        addr = struct.unpack(">I", rom[offset + 2 : offset + 6])[0]
        reg_name = SSF2_REGS.get(addr, f"${addr:08X}")
        return f"move.b  d{dn}, ({reg_name}).l", 6

    # TST.B (xxx).L -> opcode 0x4A39, 32-bit addr
    if op == 0x4A39 and offset + 6 <= len(rom):
        addr = struct.unpack(">I", rom[offset + 2 : offset + 6])[0]
        reg_name = SSF2_REGS.get(addr, f"${addr:08X}")
        return f"tst.b   ({reg_name}).l", 6

    # JSR (xxx).L -> opcode 0x4EB9, 32-bit addr
    if op == 0x4EB9 and offset + 6 <= len(rom):
        addr = struct.unpack(">I", rom[offset + 2 : offset + 6])[0]
        return f"jsr     (${addr:08X}).l", 6

    # JMP (xxx).L -> opcode 0x4EF9, 32-bit addr
    if op == 0x4EF9 and offset + 6 <= len(rom):
        addr = struct.unpack(">I", rom[offset + 2 : offset + 6])[0]
        return f"jmp     (${addr:08X}).l", 6

    # RTS -> opcode 0x4E75
    if op == 0x4E75:
        return "rts", 2

    # RTE -> opcode 0x4E73
    if op == 0x4E73:
        return "rte", 2

    return f"dc.w    ${op:04X}", 2


def scan_ssf2_accesses(rom):
    """Find every 32-bit reference to $00A130F1..$00A130FF across the entire 40Mbit ROM."""
    hits = []
    n = len(rom)
    for pos in range(0, n - 4, 2):
        val = struct.unpack(">I", rom[pos : pos + 4])[0]
        if val in SSF2_REGS:
            reg_name = SSF2_REGS[val]
            # Check preceding opcode (2 bytes or 4 bytes earlier)
            inst_offset = pos - 2
            if pos >= 4:
                prev2 = struct.unpack(">H", rom[pos - 4 : pos - 2])[0]
                if prev2 == 0x13FC:  # MOVE.B #imm, (xxx).L
                    inst_offset = pos - 4

            disasm, size = decode_simple_m68k(rom, inst_offset)
            hits.append({
                "rom_offset": f"0x{inst_offset:06X}",
                "rom_offset_int": inst_offset,
                "bank": inst_offset // BANK_SIZE,
                "bank_offset": f"0x{inst_offset % BANK_SIZE:05X}",
                "register": f"0x{val:06X}",
                "register_name": reg_name,
                "disassembly": disasm,
                "bytes": rom[inst_offset : inst_offset + size].hex().upper(),
            })
    return hits


def scan_neto_strings(rom):
    """Locate Neto's custom debug/exception strings and bank markers."""
    found = []
    for sig in NETO_SIGNATURES:
        start = 0
        while True:
            idx = rom.find(sig, start)
            if idx == -1:
                break
            # Extract up to 64 printable Latin-1 bytes for context
            end = idx
            while end < min(len(rom), idx + 64) and 0x20 <= rom[end] <= 0xFF:
                end += 1
            preview = rom[idx:end].decode("latin-1", errors="replace")
            found.append({
                "rom_offset": f"0x{idx:06X}",
                "rom_offset_int": idx,
                "bank": idx // BANK_SIZE,
                "bank_offset": f"0x{idx % BANK_SIZE:05X}",
                "signature": sig.decode("latin-1"),
                "preview": preview,
            })
            start = idx + len(sig)
    found.sort(key=lambda x: x["rom_offset_int"])
    return found


def main():
    parser = argparse.ArgumentParser(description="Locate Neto's SSF2 mapper & debug routines")
    parser.add_argument("--rom", default="Sonic_Delta_Origins_2026.bin", help="Path to 40Mbit ROM")
    parser.add_argument("--json-out", default="neto_custom_hooks.json", help="JSON report output")
    parser.add_argument("--inc-out", default="src/custom_neto/neto_hooks_map.inc", help="ASM include output")
    args = parser.parse_args()

    rom_path = Path(args.rom)
    if not rom_path.exists():
        raise FileNotFoundError(f"ROM not found: {rom_path}")

    rom = rom_path.read_bytes()
    print(f"[*] Scanning {rom_path} ({len(rom):,} bytes) for Neto's custom code...")

    ssf2_hits = scan_ssf2_accesses(rom)
    string_hits = scan_neto_strings(rom)

    report = {
        "rom": str(rom_path),
        "ssf2_access_count": len(ssf2_hits),
        "neto_string_count": len(string_hits),
        "ssf2_accesses": ssf2_hits,
        "neto_strings": string_hits,
    }

    Path(args.json_out).write_text(json.dumps(report, indent=2, ensure_ascii=False), encoding="utf-8")

    inc_path = Path(args.inc_out)
    inc_path.parent.mkdir(parents=True, exist_ok=True)
    lines = [
        "; ============================================================================",
        "; Auto-generated Map of Neto's Custom SSF2 Mapper & Debug Hooks",
        f"; Source ROM: {rom_path.name}",
        "; ============================================================================",
        "",
        "; --- SSF2 Mapper & SRAM Register Instructions ---",
    ]
    for h in ssf2_hits:
        lines.append(
            f"Neto_SSF2_Hook_{h['rom_offset_int']:06X} equ ${h['rom_offset_int']:06X} "
            f"; Bank {h['bank']:02d} @ {h['bank_offset']} -> {h['disassembly']}"
        )

    lines.extend(["", "; --- Neto Debug, Exception & Bank Marker Strings ---"])
    for s in string_hits:
        safe_preview = s["preview"].replace("\n", " ").replace("\r", "")
        lines.append(
            f"Neto_Str_{s['rom_offset_int']:06X}       equ ${s['rom_offset_int']:06X} "
            f"; Bank {s['bank']:02d} @ {s['bank_offset']} -> \"{safe_preview}\""
        )

    inc_path.write_text("\n".join(lines) + "\n", encoding="utf-8")

    print(f"[+] Found {len(ssf2_hits)} SSF2 mapper accesses and {len(string_hits)} Neto string markers.")
    print(f"[+] Wrote JSON report -> {args.json_out}")
    print(f"[+] Wrote ASM map     -> {inc_path}")


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
import_upstream_constants.py
Parses _Variables.asm, _Constants.asm, s2.constants.asm, s3.constants.asm,
and sk.constants.asm to extract Work RAM ($FF0000-$FFFFFF), Z80 RAM ($A00000-$A01FFF),
and hardware register symbols into upstream_ram_symbols.json for Ghidra.
"""

import re
import json
import argparse
from pathlib import Path

CONSTANT_FILES = [
    ("s1", "s1disasm-AS/_Constants.asm"),
    ("s1", "s1disasm-AS/_Variables.asm"),
    ("s2", "s2disasm-master/s2.constants.asm"),
    ("s3", "skdisasm-master/s3.constants.asm"),
    ("sk", "skdisasm-master/sk.constants.asm"),
]

# Matches lines like: "v_screenposx = $FFFFF700" or "Camera_X_pos equ $FFFFEE00"
DEF_RE = re.compile(
    r"^\s*([A-Za-z_][A-Za-z0-9_]*)\s*(?:=|equ|EQU)\s*\$([0-9A-Fa-f]{4,8})\b"
)


def normalize_68k_ram_addr(val):
    """
    Normalize 16-bit sign-extended RAM addresses ($FFFF8000 or $8000 -> $00FF8000)
    and 32-bit addresses into valid 24-bit Mega Drive address space.
    """
    addr = val & 0x00FFFFFF
    # If written as 16-bit short RAM ($8000..$FFFF), map to Work RAM ($FF8000..$FFFFFF)
    if val <= 0xFFFF and val >= 0x8000:
        addr = 0x00FF0000 | val
    return addr


def is_valid_hardware_or_ram(addr):
    """Keep Work RAM ($FF0000-$FFFFFF), Z80 ($A00000-$A01FFF), and I/O/VDP ($A10000-$C0001F)."""
    if 0x00FF0000 <= addr <= 0x00FFFFFF:
        return True
    if 0x00A00000 <= addr <= 0x00A01FFF:
        return True
    if 0x00A10000 <= addr <= 0x00C0001F:
        return True
    return False


def main():
    parser = argparse.ArgumentParser(description="Extract upstream RAM & hardware symbols")
    parser.add_argument("--out", default="upstream_ram_symbols.json", help="Output JSON path")
    args = parser.parse_args()

    symbols_by_addr = {}
    parsed_files = 0

    for prefix, rel_path in CONSTANT_FILES:
        p = Path(rel_path)
        if not p.exists():
            print(f"[!] Skipping missing file: {p}")
            continue

        parsed_files += 1
        text = p.read_text(encoding="utf-8", errors="ignore")
        for line in text.splitlines():
            # Strip comments
            code_part = line.split(";")[0]
            m = DEF_RE.match(code_part)
            if not m:
                continue
            name = m.group(1)
            raw_val = int(m.group(2), 16)
            addr = normalize_68k_ram_addr(raw_val)
            if not is_valid_hardware_or_ram(addr):
                continue

            addr_key = f"0x{addr:06X}"
            entry = symbols_by_addr.setdefault(addr_key, {"address_int": addr, "names": []})
            tagged_name = f"{prefix}_{name}"
            if tagged_name not in entry["names"]:
                entry["names"].append(tagged_name)

    Path(args.out).write_text(json.dumps(symbols_by_addr, indent=2), encoding="utf-8")
    print(f"[+] Parsed {parsed_files} constant files -> {len(symbols_by_addr)} unique RAM/HW addresses saved to {args.out}")


if __name__ == "__main__":
    main()

import os
import json
from ghidra.program.model.symbol import SourceType
from ghidra.program.model.data import ByteDataType, ArrayDataType
from ghidra.app.cmd.disassemble import DisassembleCommand

BANK_SIZE = 0x80000


def sanitize_symbol(name):
    clean = "".join(c if c.isalnum() else "_" for c in name)
    while "__" in clean:
        clean = clean.replace("__", "_")
    return clean.strip("_")


def label_genesis_hardware():
    """Label standard Mega Drive / Genesis and Sega SSF2 mapper registers."""
    hw_regs = {
        0x00A10001: "HW_VERSION",
        0x00A10003: "JOYPAD_1_DATA",
        0x00A10005: "JOYPAD_2_DATA",
        0x00A10009: "JOYPAD_1_CTRL",
        0x00A1000B: "JOYPAD_2_CTRL",
        0x00A11100: "Z80_BUSREQ",
        0x00A11200: "Z80_RESET",
        0x00A130F1: "SSF2_SRAM_CTRL",
        0x00A130F3: "SSF2_BANK_1",
        0x00A130F5: "SSF2_BANK_2",
        0x00A130F7: "SSF2_BANK_3",
        0x00A130F9: "SSF2_BANK_4",
        0x00A130FB: "SSF2_BANK_5",
        0x00A130FD: "SSF2_BANK_6",
        0x00A130FF: "SSF2_BANK_7",
        0x00C00000: "VDP_DATA",
        0x00C00004: "VDP_CTRL",
        0x00C00008: "VDP_HV_COUNTER",
        0x00C00011: "PSG_CTRL",
    }
    sym_table = currentProgram.getSymbolTable()
    space = currentProgram.getAddressFactory().getDefaultAddressSpace()

    for addr_int, name in hw_regs.items():

            sym_table.createLabel(space.getAddress(addr_int), name, SourceType.USER_DEFINED)
        except Exception:
            pass


def apply_ram_symbols():
    """Import community Work RAM & Z80 symbols from upstream_ram_symbols.json."""
    ram_path = "upstream_ram_symbols.json"
    if not os.path.exists(ram_path):
        print("[!] upstream_ram_symbols.json not found; skipping RAM symbol import.")
        return

    with open(ram_path, "r") as f:
        ram_map = json.load(f)

    sym_table = currentProgram.getSymbolTable()
    space = currentProgram.getAddressFactory().getDefaultAddressSpace()
    count = 0

    for _, info in ram_map.items():
        addr = space.getAddress(info["address_int"])
        names = info.get("names", [])
        if not names:
            continue
        # Combine aliases if S1/S2/S3K share the exact same RAM slot
        primary_name = sanitize_symbol("_or_".join(names[:3]))
        try:
            sym_table.createLabel(addr, primary_name, SourceType.USER_DEFINED)
            count += 1
        except Exception:
            pass

    print("[+] Imported {} Work RAM / Hardware symbols".format(count))


def apply_upstream_matches(bank_index):
    """Import symbols and block off binary assets from upstream_matches.json."""
    matches_path = "upstream_matches.json"
    if not os.path.exists(matches_path):
        print("[!] upstream_matches.json not found; skipping upstream symbol import.")
        return

    with open(matches_path, "r") as f:
        report = json.load(f)

    sym_table = currentProgram.getSymbolTable()
    listing = currentProgram.getListing()
    space = currentProgram.getAddressFactory().getDefaultAddressSpace()
    base_addr = currentProgram.getImageBase().getOffset()

    applied = 0
    for m in report.get("matches", []):
        if m.get("bank") != bank_index:
            continue

        local_offset = int(m["bank_offset"], 16)
        size = m["size"]
        kind = m["kind"]
        repo = m["repo"]
        sym_raw = m["symbol"]
        mode = m.get("match_mode", "exact")

        target_addr = space.getAddress(base_addr + local_offset)
        sym_name = "{}_{}_{:05X}".format(repo, sanitize_symbol(sym_raw), local_offset)

        try:
            sym_table.createLabel(target_addr, sym_name, SourceType.USER_DEFINED)
            comment = "[UPSTREAM: {} | {} | {} | {} bytes]".format(repo, kind, mode, size)
            listing.setComment(target_addr, 0, comment)
        except Exception as e:
            print("[!] Label error at 0x{:05X}: {}".format(local_offset, e))

        if kind == "asset":
            end_addr = space.getAddress(base_addr + local_offset + size - 1)
            listing.clearCodeUnits(target_addr, end_addr, False)
            try:
                arr_type = ArrayDataType(ByteDataType.dataType, size, 1)
                listing.createData(target_addr, arr_type)
            except Exception:
                pass
        elif kind == "code":
            cmd = DisassembleCommand(target_addr, None, True)
            cmd.applyTo(currentProgram)

        applied += 1

    print("[+] Applied {} upstream symbols/assets to Bank {:02d}".format(applied, bank_index))


def apply_neto_hooks(bank_index):
    """Label Neto's SSF2 bank-switch instructions & debug screens from neto_custom_hooks.json."""
    hooks_path = "neto_custom_hooks.json"
    if not os.path.exists(hooks_path):
        return

    with open(hooks_path, "r") as f:
        report = json.load(f)

    sym_table = currentProgram.getSymbolTable()
    listing = currentProgram.getListing()
    space = currentProgram.getAddressFactory().getDefaultAddressSpace()
    base_addr = currentProgram.getImageBase().getOffset()

    count = 0
    for h in report.get("ssf2_accesses", []):
        if h.get("bank") != bank_index:
            continue
        local_offset = int(h["bank_offset"], 16)
        target_addr = space.getAddress(base_addr + local_offset)
        lbl = "Neto_SSF2_{}_{:05X}".format(h["register_name"], local_offset)
        try:
            sym_table.createLabel(target_addr, lbl, SourceType.USER_DEFINED)
            listing.setComment(target_addr, 0, "[NETO CUSTOM: {}]".format(h["disassembly"]))
            DisassembleCommand(target_addr, None, True).applyTo(currentProgram)
            count += 1
        except Exception:
            pass

    for s in report.get("neto_strings", []):
        if s.get("bank") != bank_index:
            continue
        local_offset = int(s["bank_offset"], 16)
        target_addr = space.getAddress(base_addr + local_offset)
        lbl = "Neto_DebugStr_{:05X}".format(local_offset)
        try:
            sym_table.createLabel(target_addr, lbl, SourceType.USER_DEFINED)
            listing.setComment(target_addr, 0, "[NETO STRING: {}]".format(s["signature"]))
            count += 1
        except Exception:
            pass

    if count > 0:
        print("[+] Labeled {} Neto custom hooks/strings in Bank {:02d}".format(count, bank_index))

This upgraded splitter slices each bank_XX.asm at every matched routine and Neto hook label, generating individual .asm files grouped by provenance (src/s1disasm/, src/s2disasm/, src/skdisasm/, src/custom_neto/, and src/banks/) just like the GitHub repositories.

#!/usr/bin/env python3
"""
split_to_repo.py (v2)
Splits the 40Mbit ROM and disassembled bank ASM files into a modular
GitHub-style repository layout:
  - assets/{s1disasm,s2disasm,skdisasm}/...  (extracted binary art, maps, sound)
  - src/{s1disasm,s2disasm,skdisasm}/...     (individual matched 68K routine .asm files)
  - src/custom_neto/...                      (Neto's SSF2 mapper & Super Depurador modules)
  - src/banks/...                            (remaining unsplit bank segments)
  - main.asm & Makefile                      (one-command vasm byte-identical build)
"""

import re
import json
import argparse
from pathlib import Path

BANK_SIZE = 0x80000  # 512 KB per SSF2 bank

# Matches top-level assembly labels like "skdisasm_NemDec_01234:" or "Neto_SSF2_...:"
LABEL_LINE_RE = re.compile(r"^([A-Za-z_][A-Za-z0-9_]*)\s*:")


def sanitize_label(path_str):
    stem = Path(path_str).stem
    clean = "".join(c if c.isalnum() else "_" for c in stem)
    while "__" in clean:
        clean = clean.replace("__", "_")
    return clean.strip("_")


def classify_symbol(label_name):
    """Route a top-level assembly label into its matching repository folder."""
    for prefix in ("s1disasm", "s2disasm", "skdisasm"):
        if label_name.startswith(prefix + "_"):
            return prefix
    if label_name.startswith("Neto_"):
        return "custom_neto"
    return None


def slice_bank_asm_by_symbols(bank_file, out_root, master_includes):
    """
    Slice a disassembled bank_XX.asm file at recognized upstream/Neto symbol boundaries,
    writing standalone .asm files for each matched module and linking them via include.
    """
    bank_stem = bank_file.stem  # e.g., bank_00
    lines = bank_file.read_text(encoding="utf-8", errors="ignore").splitlines()

    segments = []  # List of (folder_tag, module_name, line_list)
    cur_folder = "banks"
    cur_name = f"{bank_stem}_header"
    cur_lines = []

    for line in lines:
        m = LABEL_LINE_RE.match(line.strip())
        if m:
            lbl = m.group(1)
            folder = classify_symbol(lbl)
            if folder:
                if cur_lines:
                    segments.append((cur_folder, cur_name, cur_lines))
                cur_folder = folder
                cur_name = lbl
                cur_lines = [line]
                continue
        cur_lines.append(line)

    if cur_lines:
        segments.append((cur_folder, cur_name, cur_lines))

    for idx, (folder, mod_name, seg_lines) in enumerate(segments):
        if folder == "banks":
            rel_path = Path("src") / "banks" / f"{bank_stem}_part{idx:03d}.asm"
        else:
            rel_path = Path("src") / folder / f"{mod_name}.asm"

        dest = out_root / rel_path
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_text("\n".join(seg_lines) + "\n", encoding="utf-8")
        master_includes.append(f'    include "{rel_path.as_posix()}"')


def build_repo_structure(rom_path, matches_json, hooks_json, asm_dir, out_repo):
    rom_data = Path(rom_path).read_bytes()
    report = json.loads(Path(matches_json).read_text(encoding="utf-8"))
    matches = report.get("matches", [])

    out = Path(out_repo)
    for sub in ("src/banks", "src/custom_neto", "src/s1disasm", "src/s2disasm", "src/skdisasm", "assets"):
        (out / sub).mkdir(parents=True, exist_ok=True)

    # 1. Extract matched upstream binary assets into organized folders
    asset_inc_lines = [
        "; ============================================================================",
        "; Auto-generated Upstream Asset Table (s1disasm / s2disasm / skdisasm)",
        "; ============================================================================",
        "",
    ]

    asset_count = 0
    for m in matches:
        if m.get("kind") != "asset":
            continue
        start = m["rom_offset_int"]
        size = m["size"]
        repo_tag = m["repo"]
        rel_upstream = Path(m["symbol"])

        dest_file = out / "assets" / repo_tag / rel_upstream
        dest_file.parent.mkdir(parents=True, exist_ok=True)
        dest_file.write_bytes(rom_data[start : start + size])

        lbl = f"{repo_tag}_{sanitize_label(str(rel_upstream))}_{start:06X}"
        rel_inc = dest_file.relative_to(out).as_posix()
        asset_inc_lines.append(f"; ROM Offset: 0x{start:06X} (Bank {m['bank']:02d} @ {m['bank_offset']})")
        asset_inc_lines.append(f"{lbl}:")
        asset_inc_lines.append(f'    incbin "{rel_inc}"')
        asset_inc_lines.append("")
        asset_count += 1

    (out / "src" / "upstream_assets.asm").write_text("\n".join(asset_inc_lines), encoding="utf-8")

    # 2. Write SSF2 Hardware Mapper & SRAM definitions
    mapper_inc = """\
; ============================================================================
; Sega SSF2 Bank Mapper & SRAM Control Registers ($A130F1 - $A130FF)
; Verified from ROM Exception Handler (Excecao $41: SRAM nao disponivel)
; ============================================================================

SRAM_CTRL       equ $A130F1   ; Write 1 to enable SRAM at $200000-$3FFFFF, 0 for ROM
BANK_REG_1      equ $A130F3   ; Maps $080000 - $0FFFFF (512 KB)
BANK_REG_2      equ $A130F5   ; Maps $100000 - $17FFFF (512 KB)
BANK_REG_3      equ $A130F7   ; Maps $180000 - $1FFFFF (512 KB)
BANK_REG_4      equ $A130F9   ; Maps $200000 - $27FFFF (512 KB)
BANK_REG_5      equ $A130FB   ; Maps $280000 - $2FFFFF (512 KB)
BANK_REG_6      equ $A130FD   ; Maps $300000 - $37FFFF (512 KB)
BANK_REG_7      equ $A130FF   ; Maps $380000 - $3FFFFF (512 KB)

SET_BANK macro reg, bank_id
    move.b  #\\bank_id, (\\reg).l
    endm
"""
    (out / "src" / "ssf2_mapper.inc").write_text(mapper_inc, encoding="utf-8")

    # 3. Copy Neto hooks map if generated by extract_neto_custom.py
    neto_inc_local = Path("src/custom_neto/neto_hooks_map.inc")
    has_neto_inc = False
    if neto_inc_local.exists():
        (out / "src" / "custom_neto" / "neto_hooks_map.inc").write_text(
            neto_inc_local.read_text(encoding="utf-8"), encoding="utf-8"
        )
        has_neto_inc = True

    # 4. Slice disassembled bank ASM files into individual routine/module files
    master_lines = [
        "; ============================================================================",
        "; Sonic Delta 40Mb (v0.79) - Reconstructed Modular M68K Disassembly",
        "; Target Assembler: vasm (vasmm68k_mot -Fbin -spaces)",
        "; ============================================================================",
        "",
        '    include "src/ssf2_mapper.inc"',
        '    include "src/upstream_assets.asm"',
    ]
    if has_neto_inc:
        master_lines.append('    include "src/custom_neto/neto_hooks_map.inc"')
    master_lines.append("")

    asm_source = Path(asm_dir)
    bank_files = sorted(asm_source.glob("bank_*.asm")) if asm_source.exists() else []
    for bf in bank_files:
        slice_bank_asm_by_symbols(bf, out, master_lines)

    (out / "main.asm").write_text("\n".join(master_lines) + "\n", encoding="utf-8")

    # 5. Write repository Makefile for vasm rebuild & SHA-256 verification
    makefile_text = f"""\
VASM = vasmm68k_mot
VASMFLAGS = -Fbin -spaces -devpac
TARGET = build/sonic_delta_40mb_rebuilt.bin
ORIG_ROM = {rom_path}

all: $(TARGET) verify

$(TARGET): main.asm
        @mkdir -p build
        $(VASM) $(VASMFLAGS) -o $(TARGET) main.asm

verify: $(TARGET)
        @echo "Comparing rebuilt binary against original ROM..."
        @cmp -l $(ORIG_ROM) $(TARGET) | head -n 20 || true
        @sha256sum $(ORIG_ROM) $(TARGET)

clean:
        rm -rf build
"""
    (out / "Makefile").write_text(makefile_text, encoding="utf-8")

    print(f"[+] Modular GitHub-style repository generated at: {out.resolve()}")
    print(f"[+] Extracted {asset_count} binary assets and sliced {len(bank_files)} bank files into routine modules.")

Append these two routines to your genesis_subroutines.asm library. Both routines follow standard Sega Mega Drive / Genesis VDP conventions (enabling DMA via register $81, computing the 68K source word-address registers $93–$97, and writing the 32-bit VDP DMA/CRAM command words).
asm
; ============================================================================
; VDP_Load_Tiles_DMA
; Transfers uncompressed 4bpp tiles (e.g., ASCII font tiles) from 68K ROM/RAM
; to VRAM using hardware 68K-to-VDP DMA.
;
; Inputs:
;   d0.l = Source 68K byte address (must be even / word-aligned)
;   d1.w = Destination VRAM byte address (e.g., $0020 for tile #1)
;   d2.w = Transfer length in words (1 tile = 32 bytes = 16 words)
; Clobbers: d0-d3, a0
; ============================================================================
VDP_Load_Tiles_DMA:
    lea     ($C00004).l, a0         ; VDP Control Port

    ; 1. Enable DMA bit in VDP Register $01 ($8174 = Display ON, VBlank ON, DMA ON)
    move.w  #$8174, (a0)

    ; 2. Set DMA Transfer Length in words (Regs $93 = low byte, $94 = high byte)
    move.w  #$9300, d3
    move.b  d2, d3
    move.w  d3, (a0)

    lsr.w   #8, d2
    move.w  #$9400, d3
    move.b  d2, d3
    move.w  d3, (a0)

    ; 3. Set DMA Source Address in words (Source >> 1 -> Regs $95, $96, $97)
    lsr.l   #1, d0                  ; Convert byte address to 23-bit word address
    move.w  #$9500, d3
    move.b  d0, d3                  ; Source low byte (A1-A8)
    move.w  d3, (a0)

    lsr.l   #8, d0
    move.w  #$9600, d3
    move.b  d0, d3                  ; Source mid byte (A9-A16)
    move.w  d3, (a0)

    lsr.l   #8, d0
    andi.b  #$7F, d0                ; Mask bit 7 = 0 (68K-to-VDP DMA mode)
    move.w  #$9700, d3
    move.b  d0, d3                  ; Source high byte (A17-A23)
    move.w  d3, (a0)

    ; 4. Build VDP VRAM DMA Command Word:
    ;    First word:  $4000 | (VRAM_addr & $3FFF)
    ;    Second word: $0080 | (VRAM_addr >> 14)   [Bit 7 triggers DMA]
    move.w  d1, d3
    andi.w  #$3FFF, d3
    ori.w   #$4000, d3
    move.w  d3, (a0)                ; Write first command word

    move.w  d1, d3
    rol.w   #2, d3
    andi.w  #$0003, d3
    ori.w   #$0080, d3
    move.w  d3, (a0)                ; Write second command word (triggers DMA)

    ; 5. Disable DMA bit in VDP Register $01 ($8164) after transfer completes
    move.w  #$8164, (a0)
    rts


; ============================================================================
; VDP_Load_Palette_CRAM
; Loads a 16-color (32-byte) or 64-color (128-byte) palette into CRAM.
;
; Inputs:
;   a1   = Source address of palette data (array of 9-bit $0BGR words)
;   d0.w = CRAM byte offset ($00 = Line 0, $20 = Line 1, $40 = Line 2, $60 = Line 3)
;   d1.w = Number of colors to copy minus 1 (e.g., #15 for one 16-color line)
; Clobbers: d0-d1, a0-a1
; ============================================================================
VDP_Load_Palette_CRAM:
    lea     ($C00004).l, a0         ; VDP Control Port

    ; Build CRAM Write Command: $C0000000 | ((CRAM_offset & $007F) << 16)
    andi.l  #$0000007F, d0
    swap    d0
    ori.l   #$C0000000, d0
    move.l  d0, (a0)                ; Set VDP auto-increment target to CRAM

    lea     ($C00000).l, a0         ; VDP Data Port
.copy_loop:
    move.w  (a1)+, (a0)             ; Write 16-bit color word ($0BBB_GGG_RRR)
    dbf     d1, .copy_loop
    rts

Yes, it is completely cross-referenceable. Because every matched routine and asset in upstream_matches.json is tied directly to its exact source path and symbol name in s1disasm-AS, s2disasm-master, and skdisasm-master, you do not have to settle for raw Ghidra disassembly output (move.w #$8004, ($C00004).l or loc_001234) for code that already exists upstream.
How Cross-Referencing Works Across the Two Codebases
    • Source-Level Substitution for Matched Routines: When match_upstream.py finds an exact match (or a relocated match where only the 32-bit JSR/JMP/LEA targets changed), you already have the original human-annotated .asm source code sitting inside s1disasm-AS, s2disasm-master, and skdisasm-master—complete with community comments, local labels (.loop, .done), RAM constant names (v_player, Camera_X_pos), and macros (ObjectMove).
    • Automatic Cross-Reference Headers (XREF): Even when keeping raw Ghidra output for modified routines, a post-processor can inject direct file-and-line citations (for example, ; Upstream Ref: s1disasm-AS/_inc/Nemesis Decompression.asm:14) and swap raw hex addresses ($FFFFF700, $C00004) with the official constant names from upstream_ram_symbols.json.
    • Call-Graph Label Propagation: Whenever a routine in src/custom_neto/ or an unmatched bank calls (jsr / bsr) an address that matched NemDec or PlaySound, a cross-reference pass replaces jsr ($00001420).l with jsr NemDec across the entire repository.

xref_enricher.py
Cross-references your split disassembly repository against s1disasm-AS,
s2disasm-master, and skdisasm-master:

  2. Replaces raw hex addresses in disassembled .asm files with human-readable symbols
  3. Annotates matched routines with exact file paths & line numbers in the upstream repos

from pathlib import Path

UPSTREAM_ROOTS = {
    "s1disasm": Path("s1disasm-AS"),
    "s2disasm": Path("s2disasm-master"),
    "skdisasm": Path("skdisasm-master"),
}

HEX_OPERAND_RE = re.compile(r"\(\$([0-9A-Fa-f]{4,8})\)\.(w|l)")


def sanitize_symbol(name):
    clean = "".join(c if c.isalnum() else "_" for c in name)
    while "__" in clean:
        clean = clean.replace("__", "_")
    return clean.strip("_")


def locate_upstream_source(repo_tag, symbol_name):
    """Find the exact .asm file and line number where a matched symbol is defined upstream."""
    root = UPSTREAM_ROOTS.get(repo_tag)
    if not root or not root.exists():
        return None

    # Strip auto-appended offset suffix if present (e.g., skdisasm_NemDec_01234 -> NemDec)
    raw_sym = symbol_name
    for prefix in ("s1disasm_", "s2disasm_", "skdisasm_"):
        if raw_sym.startswith(prefix):
            raw_sym = raw_sym[len(prefix):]
    raw_sym = re.sub(r"_[0-9A-Fa-f]{5,6}$", "", raw_sym)

    pattern = re.compile(rf"^\s*{re.escape(raw_sym)}\s*:", re.MULTILINE)
    for asm_file in root.rglob("*.asm"):
        try:
            text = asm_file.read_text(encoding="utf-8", errors="ignore")
        except Exception:
            continue
        m = pattern.search(text)
        if m:
            line_no = text[: m.start()].count("\n") + 1
            return f"{asm_file.as_posix()}:{line_no} ({raw_sym})"
    return None


def build_symbol_tables(matches_path, ram_path, hooks_path):
    """Combine ROM labels, RAM variables, and Neto custom hooks into lookup dictionaries."""
    addr_to_label = {}
    const_defs = {}

    # 1. Hardware registers
    hw = {
        0x00A10001: "HW_VERSION",
        0x00A10003: "JOYPAD_1_DATA",
        0x00A10005: "JOYPAD_2_DATA",
        0x00A10009: "JOYPAD_1_CTRL",
        0x00A1000B: "JOYPAD_2_CTRL",
        0x00A11100: "Z80_BUSREQ",
        0x00A11200: "Z80_RESET",
        0x00C00000: "VDP_DATA",
        0x00C00004: "VDP_CTRL",
        0x00C00008: "VDP_HV_COUNTER",
        0x00C00011: "PSG_CTRL",
    }
    for addr, name in hw.items():
        addr_to_label[addr] = name
        const_defs[name] = f"${addr:08X}"

    # 2. Upstream Work RAM & Z80 symbols
    if Path(ram_path).exists():
        ram_data = json.loads(Path(ram_path).read_text(encoding="utf-8"))
        for _, info in ram_data.items():
            addr = info["address_int"]
            names = info.get("names", [])
            if names:
                lbl = sanitize_symbol(names[0])
                addr_to_label[addr] = lbl
                const_defs[lbl] = f"${addr:08X}"

    # 3. Matched upstream ROM routines & assets
    if Path(matches_path).exists():
        match_data = json.loads(Path(matches_path).read_text(encoding="utf-8"))
        for m in match_data.get("matches", []):
            addr = m["rom_offset_int"]
            lbl = f"{m['repo']}_{sanitize_symbol(m['symbol'])}_{addr:06X}"
            addr_to_label[addr] = lbl
            const_defs[lbl] = f"${addr:08X}"

    # 4. Neto custom SSF2 & debug hooks
    if Path(hooks_path).exists():
        hook_data = json.loads(Path(hooks_path).read_text(encoding="utf-8"))
        for h in hook_data.get("ssf2_accesses", []):
            addr = h["rom_offset_int"]
            lbl = f"Neto_SSF2_{h['register_name']}_{addr:06X}"
            addr_to_label[addr] = lbl
            const_defs[lbl] = f"${addr:08X}"

    return addr_to_label, const_defs


def enrich_repository(repo_dir, addr_to_label, const_defs):
    repo = Path(repo_dir)
    src_dir = repo / "src"
    if not src_dir.exists():
        raise FileNotFoundError(f"Repository src/ folder not found at {src_dir}")

    # Write human_constants.inc so vasm resolves every substituted label
    inc_lines = [
        "; ============================================================================",
        "; Auto-generated Cross-Referenced Human-Readable Symbols (S1 / S2 / S3K / Neto)",
        "; ============================================================================",
        "",
    ]
    for name, hex_val in sorted(const_defs.items()):
        inc_lines.append(f"{name:<44} equ {hex_val}")

    inc_path = src_dir / "human_constants.inc"
    inc_path.write_text("\n".join(inc_lines) + "\n", encoding="utf-8")

    # Prepend human_constants.inc to main.asm if not already included
    main_asm = repo / "main.asm"
    if main_asm.exists():
        main_txt = main_asm.read_text(encoding="utf-8")
        inc_stmt = '    include "src/human_constants.inc"'
        if inc_stmt not in main_txt:
            main_asm.write_text(inc_stmt + "\n" + main_txt, encoding="utf-8")

    # Rewrite raw hex addresses and inject upstream file:line citations in all .asm files
    rewritten_files = 0
    replaced_refs = 0

    for asm_file in src_dir.rglob("*.asm"):
        if asm_file.name in ("upstream_assets.asm",):
            continue

        lines = asm_file.read_text(encoding="utf-8", errors="ignore").splitlines()
        new_lines = []
        changed = False

        # Add upstream source cross-reference banner at top of matched routine files
        stem = asm_file.stem
        for repo_tag in ("s1disasm", "s2disasm", "skdisasm"):
            if stem.startswith(repo_tag + "_"):
                xref_loc = locate_upstream_source(repo_tag, stem)
                if xref_loc and (not lines or "XREF Upstream Source:" not in lines[0]):
                    new_lines.append(f"; XREF Upstream Source: {xref_loc}")
                    changed = True
                break

        for line in lines:
            def _sub_addr(match):
                nonlocal replaced_refs, changed
                raw_hex = match.group(1)
                size_suffix = match.group(2)
                val = int(raw_hex, 16)
                if size_suffix == "w" and val >= 0x8000:
                    val = 0x00FF0000 | val
                lbl = addr_to_label.get(val)
                if lbl:
                    replaced_refs += 1
                    changed = True
                    return f"({lbl}).{size_suffix}"
                return match.group(0)

            updated_line = HEX_OPERAND_RE.sub(_sub_addr, line)
            new_lines.append(updated_line)

        if changed:
            asm_file.write_text("\n".join(new_lines) + "\n", encoding="utf-8")
            rewritten_files += 1

    print(f"[+] Wrote {len(const_defs)} human-readable constants -> {inc_path}")
    print(f"[+] Cross-referenced {replaced_refs} hex operands across {rewritten_files} .asm files.")


def main():
    parser = argparse.ArgumentParser(description="Enrich split disassembly with human-readable cross-references")
    parser.add_argument("--repo", default="sonic_delta_disasm_repo", help="Generated repository folder")
    parser.add_argument("--matches", default="upstream_matches.json", help="Upstream matches JSON")

    addr_to_label, const_defs = build_symbol_tables(args.matches, args.ram, args.hooks)

    main()

#!/usr/bin/env python3
"""
macro_beautifier.py
Generates src/sonic_macros.inc (modeled on s1disasm/s2disasm/skdisasm macros)
and converts raw hardware/engine instruction sequences in src/*.asm into
human-readable macros while preserving byte-identical vasm output.
"""

import re
import argparse
from pathlib import Path

SONIC_MACROS_INC = """\
; ============================================================================
; Community-Style Sonic Mega Drive / Genesis Macros (vasm Motorola syntax)
; ============================================================================

; Write a 32-bit VDP VRAM/CRAM/VSRAM command longword to VDP_CTRL
writeVRAM macro addr
    move.l  #(($40000000|((\\addr)&$3FFF)<<16)|((\\addr)>>14)), (VDP_CTRL).l
    endm

writeCRAM macro addr
    move.l  #(($C0000000|((\\addr)&$3FFF)<<16)|((\\addr)>>14)), (VDP_CTRL).l
    endm

writeVSRAM macro addr
    move.l  #(($40000010|((\\addr)&$3FFF)<<16)|((\\addr)>>14)), (VDP_CTRL).l
    endm

; Set a single VDP register ($80xx - $97xx)
setVDPReg macro reg, val
    move.w  #((($80|(\\reg))<<8)|(\\val)), (VDP_CTRL).l
    endm

; Request / Release Z80 Bus ($A11100)
stopZ80 macro
    move.w  #$0100, (Z80_BUSREQ).l
    endm

startZ80 macro
    move.w  #$0000, (Z80_BUSREQ).l
    endm
"""

# Regex patterns matching raw 68K instructions after xref_enricher.py has run
VDP_LONG_RE = re.compile(
    r"^(\s*)move\.l\s+#\$([0-9A-Fa-f]{8})\s*,\s*\(VDP_CTRL\)\.l\b(.*)$"
)
VDP_REG_RE = re.compile(
    r"^(\s*)move\.w\s+#\$8([0-9A-Fa-f])([0-9A-Fa-f]{2})\s*,\s*\(VDP_CTRL\)\.l\b(.*)$"
)
Z80_STOP_RE = re.compile(
    r"^(\s*)move\.w\s+#\$(?:0100|100)\s*,\s*\(Z80_BUSREQ\)\.l\b(.*)$"
)
Z80_START_RE = re.compile(
    r"^(\s*)move\.w\s+#\$(?:0000|0)\s*,\s*\(Z80_BUSREQ\)\.l\b(.*)$"
)


def decode_vdp_cmd_long(cmd_val):
    """Decode a 32-bit VDP control port command into (macro_name, target_offset)."""
    cd = ((cmd_val >> 30) & 0x03) | (((cmd_val >> 4) & 0x0F) << 2)
    addr = ((cmd_val >> 16) & 0x3FFF) | ((cmd_val & 0x0003) << 14)
    # Only convert non-DMA writes (bit 7 of low word == 0)
    if (cmd_val & 0x0080) != 0:
        return None
    if cd == 0x01:
        return ("writeVRAM", addr)
    if cd == 0x03:
        return ("writeCRAM", addr)
    if cd == 0x05:
        return ("writeVSRAM", addr)
    return None


def beautify_line(line):
    m = VDP_LONG_RE.match(line)
    if m:
        indent, hex_cmd, trailing = m.groups()
        decoded = decode_vdp_cmd_long(int(hex_cmd, 16))
        if decoded:
            macro_name, target_addr = decoded
            return f"{indent}{macro_name} ${target_addr:04X}{trailing}", True

    m = VDP_REG_RE.match(line)
    if m:
        indent, reg_nib, val_hex, trailing = m.groups()
        reg_num = int(reg_nib, 16)
        return f"{indent}setVDPReg ${reg_num:02X}, ${val_hex}{trailing}", True

    m = Z80_STOP_RE.match(line)
    if m:
        indent, trailing = m.groups()
        return f"{indent}stopZ80{trailing}", True

    m = Z80_START_RE.match(line)
    if m:
        indent, trailing = m.groups()
        return f"{indent}startZ80{trailing}", True

    return line, False


def main():
    parser = argparse.ArgumentParser(description="Convert raw VDP/Z80 writes into Sonic community macros")
    parser.add_argument("--repo", default="sonic_delta_disasm_repo", help="Generated repository folder")
    args = parser.parse_args()

    repo = Path(args.repo)
    src_dir = repo / "src"
    if not src_dir.exists():
        raise FileNotFoundError(f"Repository src/ folder not found at {src_dir}")

    # 1. Write src/sonic_macros.inc
    macro_inc = src_dir / "sonic_macros.inc"
    macro_inc.write_text(SONIC_MACROS_INC, encoding="utf-8")

    # 2. Include sonic_macros.inc in main.asm right after human_constants.inc
    main_asm = repo / "main.asm"
    if main_asm.exists():
        main_txt = main_asm.read_text(encoding="utf-8")
        inc_stmt = '    include "src/sonic_macros.inc"'
        if inc_stmt not in main_txt:
            main_asm.write_text(inc_stmt + "\n" + main_txt, encoding="utf-8")

    # 3. Rewrite matching lines across src/**/*.asm
    files_changed = 0
    macros_applied = 0
    for asm_file in src_dir.rglob("*.asm"):
        if asm_file.name == "upstream_assets.asm":
            continue
        lines = asm_file.read_text(encoding="utf-8", errors="ignore").splitlines()
        out_lines = []
        file_modified = False
        for line in lines:
            new_line, changed = beautify_line(line)
            out_lines.append(new_line)
            if changed:
                file_modified = True
                macros_applied += 1
        if file_modified:
            asm_file.write_text("\n".join(out_lines) + "\n", encoding="utf-8")
            files_changed += 1

    print(f"[+] Wrote community macro library -> {macro_inc}")
    print(f"[+] Applied {macros_applied} macro substitutions across {files_changed} .asm files.")


if __name__ == "__main__":
    main()

out:flush()

local REG_NAMES = {
    [0xA130F1] = { "SSF2_SRAM_CTRL", "SRAM/ROM Toggle ($200000-$3FFFFF)" },
    [0xA130F3] = { "SSF2_BANK_1",    "Slot 1 ($080000-$0FFFFF)" },
    [0xA130F5] = { "SSF2_BANK_2",    "Slot 2 ($100000-$17FFFF)" },
    [0xA130F7] = { "SSF2_BANK_3",    "Slot 3 ($180000-$1FFFFF)" },
    [0xA130F9] = { "SSF2_BANK_4",    "Slot 4 ($200000-$27FFFF)" },
    [0xA130FB] = { "SSF2_BANK_5",    "Slot 5 ($280000-$2FFFFF)" },
    [0xA130FD] = { "SSF2_BANK_6",    "Slot 6 ($300000-$37FFFF)" },
    [0xA130FF] = { "SSF2_BANK_7",    "Slot 7 ($380000-$3FFFFF)" },
}

local function log_write(pc, addr, val)
    local info = REG_NAMES[addr]
    if not info then return end
    local line = string.format("0x%06X | %-14s | 0x%02X | %s\n", pc & 0xFFFFFF, info[1], val & 0xFF, info[2])
    out:write(line)
    out:flush()
    print("[SSF2] " .. line:gsub("\n", ""))
end

-- Branch 1: Mesen API
if emu and emu.addMemoryCallback then
    print("[*] Initializing SSF2 Watchpoints in Mesen...")
    emu.addMemoryCallback(function(addr, val)
        local state = emu.getState()
        local pc = state["cpu.pc"] or state["pc"] or 0
        log_write(pc, addr, val)
    end, emu.callbackType.write, 0xA130F1, 0xA130FF)

-- Branch 2: MAME API (mame genesis -autoboot_script trace_ssf2_live.lua ...)
elseif manager and manager.machine then
    print("[*] Initializing SSF2 Watchpoints in MAME...")
    local cpu = manager.machine.devices[":maincpu"]
    local space = cpu.spaces["program"]
    space:install_write_tap(0xA130F1, 0xA130FF, "ssf2_mapper_tap", function(offset, data, mask)
        local pc = cpu.state["PC"].value
        log_write(pc, offset, data)
        return data
    end)
else
    print("[!] Unsupported Lua host environment.")
end

#!/usr/bin/env python3
"""
collect_trace.py (v2)
Parses live SSF2 mapper watchpoint traces (trace_banks.txt) produced by:
  - trace_ssf2_live.lua (MAME / Mesen)
  - blastem_trace_ssf2.gdb (BlastEm GDB stub)
Deduplicates repeated writes, resolves target physical ROM offsets, and
outputs resolved_switches.json for Ghidra and repository cross-referencing.
"""

import re
import json
import argparse
from pathlib import Path

BANK_SIZE = 0x80000  # 512 KB per SSF2 bank

SLOT_WINDOWS = {
    "SSF2_SRAM_CTRL": ("SRAM/ROM Control", 0x200000),
    "SSF2_BANK_1":    ("Slot 1", 0x080000),
    "SSF2_BANK_2":    ("Slot 2", 0x100000),
    "SSF2_BANK_3":    ("Slot 3", 0x180000),
    "SSF2_BANK_4":    ("Slot 4", 0x200000),
    "SSF2_BANK_5":    ("Slot 5", 0x280000),
    "SSF2_BANK_6":    ("Slot 6", 0x300000),
    "SSF2_BANK_7":    ("Slot 7", 0x380000),
}

# Matches: "0x000408 | SSF2_BANK_1    | 0x04 | Slot 1 ($080000-$0FFFFF)"
TRACE_LINE_RE = re.compile(
    r"^\s*0x([0-9A-Fa-f]+)\s*\|\s*([A-Z0-9_]+)\s*\|\s*0x([0-9A-Fa-f]+)\s*(?:\|\s*(.*))?$"
)


def parse_trace_file(trace_path: Path):
    """Parse raw trace_banks.txt lines into structured switch events."""
    if not trace_path.exists():
        raise FileNotFoundError(f"Trace log not found: {trace_path}")

    events = []
    seen_pairs = {}  # (pc_int, reg_name, val_int) -> count

    for line_no, raw_line in enumerate(trace_path.read_text(encoding="utf-8", errors="ignore").splitlines(), 1):
        line = raw_line.strip()
        if not line or line.startswith("#") or line.startswith("Hardware watchpoint"):
            continue

        m = TRACE_LINE_RE.match(line)
        if not m:
            continue

        pc_val = int(m.group(1), 16) & 0xFFFFFF
        reg_name = m.group(2).strip()
        bank_val = int(m.group(3), 16) & 0xFF
        note = (m.group(4) or "").strip()

        key = (pc_val, reg_name, bank_val)
        seen_pairs[key] = seen_pairs.get(key, 0) + 1

        if seen_pairs[key] == 1:
            slot_label, base_vaddr = SLOT_WINDOWS.get(reg_name, ("Unknown Slot", 0))
            target_rom_offset = None
            if reg_name != "SSF2_SRAM_CTRL":
                target_rom_offset = f"0x{bank_val * BANK_SIZE:06X}"

            events.append({
                "pc": f"0x{pc_val:06X}",
                "pc_int": pc_val,
                "pc_bank": pc_val // BANK_SIZE,
                "pc_bank_offset": f"0x{pc_val % BANK_SIZE:05X}",
                "register": reg_name,
                "slot": slot_label,
                "mapped_window_start": f"0x{base_vaddr:06X}",
                "value_written": f"0x{bank_val:02X}",
                "value_int": bank_val,
                "target_rom_offset": target_rom_offset,
                "note": note,
                "hit_count": 1,
            })

    # Attach exact hit counts to deduplicated records
    for ev in events:
        key = (ev["pc_int"], ev["register"], ev["value_int"])
        ev["hit_count"] = seen_pairs[key]

    return events


def main():
    parser = argparse.ArgumentParser(description="Parse MAME/Mesen/BlastEm SSF2 trace logs")
    parser.add_argument("--trace", default="trace_banks.txt", help="Input trace log path")
    parser.add_argument("--out", default="resolved_switches.json", help="Output JSON path")
    args = parser.parse_args()

    events = parse_trace_file(Path(args.trace))

    # Group observed bank IDs per slot for quick summary
    slot_summary = {}
    for ev in events:
        reg = ev["register"]
        slot_summary.setdefault(reg, set()).add(ev["value_written"])

    report = {
        "source_trace": args.trace,
        "unique_switch_sites": len(events),
        "total_logged_writes": sum(e["hit_count"] for e in events),
        "observed_values_per_register": {
            k: sorted(v) for k, v in sorted(slot_summary.items())
        },
        "switches": events,
    }

    Path(args.out).write_text(json.dumps(report, indent=2), encoding="utf-8")
    print(f"[+] Parsed {report['total_logged_writes']} total SSF2 writes across {len(events)} unique PCs -> {args.out}")
    for reg, vals in sorted(slot_summary.items()):
        print(f"    {reg:<16}: observed values {', '.join(sorted(vals))}")


if __name__ == "__main__":
    main()

    switches_path = "resolved_switches.json"

            listing.setComment(target_addr, 0, comment)

        except Exception:
            pass

#!/usr/bin/env python3
"""
xref_enricher.py (v2)
Cross-references your split disassembly repository against:
  1. s1disasm-AS, s2disasm-master, and skdisasm-master (symbols, RAM constants, file:line XREFs)
  2. neto_custom_hooks.json (static SSF2 & debug hook labels)
  3. resolved_switches.json (live BlastEm/MAME/Mesen runtime bank-switch traces)
Outputs:
  - src/human_constants.inc
  - src/custom_neto/runtime_bank_switches.inc
  - Inline symbol substitutions & runtime bank-switch comments across src/**/*.asm
"""

import re
import json
import argparse
from pathlib import Path

UPSTREAM_ROOTS = {
    "s1disasm": Path("s1disasm-AS"),
    "s2disasm": Path("s2disasm-master"),
    "skdisasm": Path("skdisasm-master"),
}

HEX_OPERAND_RE = re.compile(r"\(\$([0-9A-Fa-f]{4,8})\)\.(w|l)")
NETO_HOOK_LABEL_RE = re.compile(r"^Neto_SSF2_[A-Z0-9_]+_([0-9A-Fa-f]{6})\s*:")


def sanitize_symbol(name):
    clean = "".join(c if c.isalnum() else "_" for c in name)
    while "__" in clean:
        clean = clean.replace("__", "_")
    return clean.strip("_")


def locate_upstream_source(repo_tag, symbol_name):
    """Find the exact .asm file and line number where a matched symbol is defined upstream."""
    root = UPSTREAM_ROOTS.get(repo_tag)
    if not root or not root.exists():
        return None

    raw_sym = symbol_name
    for prefix in ("s1disasm_", "s2disasm_", "skdisasm_"):
        if raw_sym.startswith(prefix):
            raw_sym = raw_sym[len(prefix):]
    raw_sym = re.sub(r"_[0-9A-Fa-f]{5,6}$", "", raw_sym)

    pattern = re.compile(rf"^\s*{re.escape(raw_sym)}\s*:", re.MULTILINE)
    for asm_file in root.rglob("*.asm"):
        try:
            text = asm_file.read_text(encoding="utf-8", errors="ignore")
        except Exception:
            continue
        m = pattern.search(text)
        if m:
            line_no = text[: m.start()].count("\n") + 1
            return f"{asm_file.as_posix()}:{line_no} ({raw_sym})"
    return None


def load_runtime_switches(switches_path):
    """Load resolved_switches.json keyed by PC integer address."""
    by_pc = {}
    p = Path(switches_path)
    if not p.exists():
        return by_pc

    data = json.loads(p.read_text(encoding="utf-8"))
    for sw in data.get("switches", []):
        pc_int = sw["pc_int"]
        by_pc.setdefault(pc_int, []).append(sw)
    return by_pc


def format_runtime_comment(sw_list):
    """Format observed runtime bank values and target ROM offsets into a concise comment."""
    parts = []
    for sw in sw_list:
        val = sw["value_written"]
        hits = sw["hit_count"]
        target = sw.get("target_rom_offset")
        if target:
            parts.append(f"{sw['register']}<-{val} (ROM {target}, {hits}x)")
        else:
            parts.append(f"{sw['register']}<-{val} ({hits}x)")
    return "; [RUNTIME TRACE: " + " | ".join(parts) + "]"


def build_symbol_tables(matches_path, ram_path, hooks_path, switches_by_pc):
    """Combine ROM labels, RAM variables, Neto hooks, and runtime trace PCs."""
    addr_to_label = {}
    const_defs = {}

    # 1. Hardware registers
    hw = {
        0x00A10001: "HW_VERSION",
        0x00A10003: "JOYPAD_1_DATA",
        0x00A10005: "JOYPAD_2_DATA",
        0x00A10009: "JOYPAD_1_CTRL",
        0x00A1000B: "JOYPAD_2_CTRL",
        0x00A11100: "Z80_BUSREQ",
        0x00A11200: "Z80_RESET",
        0x00C00000: "VDP_DATA",
        0x00C00004: "VDP_CTRL",
        0x00C00008: "VDP_HV_COUNTER",
        0x00C00011: "PSG_CTRL",
    }
    for addr, name in hw.items():
        addr_to_label[addr] = name
        const_defs[name] = f"${addr:08X}"

    # 2. Upstream Work RAM & Z80 symbols
    if Path(ram_path).exists():
        ram_data = json.loads(Path(ram_path).read_text(encoding="utf-8"))
        for _, info in ram_data.items():
            addr = info["address_int"]
            names = info.get("names", [])
            if names:
                lbl = sanitize_symbol(names)
                addr_to_label[addr] = lbl
                const_defs[lbl] = f"${addr:08X}"

    # 3. Matched upstream ROM routines & assets
    if Path(matches_path).exists():
        match_data = json.loads(Path(matches_path).read_text(encoding="utf-8"))
        for m in match_data.get("matches", []):

#!/usr/bin/env python3
"""
source_block_swapper.py
Replaces exact-match 68000 routines in sonic_delta_disasm_repo/src/ with the
verbatim community-commented .asm source blocks (including original local
labels like .loop/.done and inline comments) from s1disasm-AS, s2disasm-master,
and skdisasm-master.

Safety guarantees:
  1. Only swaps routines with match_mode == "exact" (never relocated routines).
  2. Preserves the original Ghidra/vasm output as a backup (.bak) or fallback
     if external branch targets are missing from human_constants.inc.
"""

import re
import json
import argparse
from pathlib import Path

UPSTREAM_ROOTS = {
    "s1disasm": Path("s1disasm-AS"),
    "s2disasm": Path("s2disasm-master"),
    "skdisasm": Path("skdisasm-master"),
}

GLOBAL_LABEL_RE = re.compile(r"^([A-Za-z_][A-Za-z0-9_]*)\s*:")
TERMINATOR_RE = re.compile(r"^\s*(?:rts|rte|jmp)\b", re.IGNORECASE)
BRANCH_TARGET_RE = re.compile(
    r"^\s*(?:bra|bsr|beq|bne|bpl|bmi|bcc|bcs|bvc|bvs|bhi|bls|bge|blt|bgt|ble|jmp|jsr)(?:\.[bws])?\s+([A-Za-z_][A-Za-z0-9_]*)",
    re.IGNORECASE,
)


def load_defined_constants(repo_dir: Path):
    """Collect all symbols defined in human_constants.inc and ssf2_mapper.inc."""
    defined = set()
    for inc_name in ("human_constants.inc", "ssf2_mapper.inc"):
        inc_file = repo_dir / "src" / inc_name
        if not inc_file.exists():
            continue
        for line in inc_file.read_text(encoding="utf-8", errors="ignore").splitlines():
            parts = line.strip().split()
            if len(parts) >= 3 and parts[1].lower() in ("equ", "="):
                defined.add(parts)
    return defined


def extract_upstream_routine(repo_tag: str, raw_symbol: str):
    """
    Search the upstream repository for `raw_symbol:` and extract the full
    annotated routine block up to the next global label after an rts/rte/jmp.
    """
    root = UPSTREAM_ROOTS.get(repo_tag)
    if not root or not root.exists():
        return None, None

    label_re = re.compile(rf"^\s*{re.escape(raw_symbol)}\s*:")
    for asm_file in root.rglob("*.asm"):
        try:
            lines = asm_file.read_text(encoding="utf-8", errors="ignore").splitlines()
        except Exception:
            continue

        for idx, line in enumerate(lines):
            if not label_re.match(line):
                continue

            # Include up to 5 preceding comment lines describing the function
            start_idx = idx
            while start_idx > 0 and idx - start_idx <= 5:
                prev = lines[start_idx - 1].strip()
                if prev.startswith(";"):
                    start_idx -= 1
                else:
                    break

            block = []
            saw_terminator = False
            for cur_idx in range(start_idx, len(lines)):
                cur_line = lines[cur_idx]
                stripped = cur_line.strip()

                # Stop at the next top-level global label once a terminator has been seen
                if cur_idx > idx:
                    m_lbl = GLOBAL_LABEL_RE.match(stripped)
                    if m_lbl and not m_lbl.group(1).startswith(("loc_", "byte_", "word_")):
                        if saw_terminator:
                            break
                        # If it falls through into another global routine without rts, abort swap
                        return None, None

                block.append(cur_line)
                if TERMINATOR_RE.match(stripped):
                    saw_terminator = True

            if saw_terminator and len(block) >= 2:
                rel_src = f"{asm_file.as_posix()}:{idx + 1}"
                return block, rel_src

    return None, None


def is_block_self_contained(block_lines, entry_symbol: str, defined_symbols: set):
    """
    Ensure every non-local branch/jump target inside the extracted upstream block
    is either the entry symbol itself or already defined in human_constants.inc.
    """
    local_labels = {entry_symbol}
    for line in block_lines:
        m_lbl = GLOBAL_LABEL_RE.match(line.strip())
        if m_lbl:
            local_labels.add(m_lbl.group(1))

    for line in block_lines:
        code_only = line.split(";")
        m_br = BRANCH_TARGET_RE.match(code_only)
        if m_br:
            target = m_br.group(1)
            # Local labels starting with '.' or '@' are safe
            if target.startswith((".", "@")):
                continue
            if target not in local_labels and target not in defined_symbols:
                return False
    return True


def main():
    parser = argparse.ArgumentParser(description="Swap exact-match routines with verbatim upstream ASM blocks")
    parser.add_argument("--repo", default="sonic_delta_disasm_repo", help="Generated repository path")
    parser.add_argument("--matches", default="upstream_matches.json", help="Upstream matches JSON")
    args = parser.parse_args()

    repo = Path(args.repo)
    matches_path = Path(args.matches)
    if not repo.exists() or not matches_path.exists():
        raise FileNotFoundError("Repository directory or upstream_matches.json not found.")

    report = json.loads(matches_path.read_text(encoding="utf-8"))
    defined_symbols = load_defined_constants(repo)

    swapped = 0
    skipped = 0

    for m in report.get("matches", []):
        if m.get("kind") != "code" or m.get("match_mode") != "exact":
            continue

        repo_tag = m["repo"]
        raw_sym = m["symbol"]
        rom_offset = m["rom_offset_int"]
        module_stem = f"{repo_tag}_{raw_sym}_{rom_offset:06X}"
        target_asm = repo / "src" / repo_tag / f"{module_stem}.asm"

        if not target_asm.exists():
            continue

        block_lines, src_loc = extract_upstream_routine(repo_tag, raw_sym)
        if not block_lines:
            skipped += 1
            continue

        if not is_block_self_contained(block_lines, raw_sym, defined_symbols):
            skipped += 1
            continue

        # Backup the Ghidra-disassembled file before swapping
        bak_file = target_asm.with_suffix(".asm.ghidra.bak")
        if not bak_file.exists():
            bak_file.write_text(target_asm.read_text(encoding="utf-8"), encoding="utf-8")

        header = [
            f"; ============================================================================",
            f"; VERBATIM UPSTREAM SOURCE SWAP: {src_loc}",
            f"; ROM Offset: 0x{rom_offset:06X} (Bank {m['bank']:02d} @ {m['bank_offset']})",
            f"; ============================================================================",
            f"{module_stem}:",
        ]
        target_asm.write_text("\n".join(header + block_lines) + "\n", encoding="utf-8")
        swapped += 1

    print(f"[+] Source-block swapper complete: {swapped} routines upgraded to verbatim community ASM ({skipped} kept as Ghidra disassembly).")


if __name__ == "__main__":
    main()
